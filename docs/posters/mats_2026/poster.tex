\postersec{TL;DR}
\begin{itemize}
  \item A single \textbf{linear map} predicts an LLM's mean answer activation from its final context activation \textbf{before generation}: held-out $R^2$ \textbf{0.75}, and rank-1 retrieval of the true answer among 10,000 candidates at \textbf{0.98} 
  \item The base model already reaches \textbf{67\% of the fully post-trained $R^2$}; SFT closes most of the rest, and DPO and RLVR barely change it
  \item It is \textbf{shared between the assistant in chat template, assistant in stories, and other fictional characters in stories}, providing evidence for the persona selection model
  \item It can be used to predict \textbf{sycophancy, hallucination and evil before inference}.
\end{itemize}

\postersec{Motivation}
\begin{itemize}
  \item LLMs can be viewed as deterministic maps from \emph{contexts} to \emph{answer distributions}
  \item A \textbf{simple approximation} of that map would be extremely useful: much easier to analyze theoretically, sample from, and understand the effects of finetuning
  \item But it need not exist: the map is billions of parameters applying dozens of nonlinear layers. 
  \item Prior work either uses simple but \emph{narrow} predictors (specific tokens, activations, behaviors), or distillation, a whole-distribution predictor which is just as complex as the original model
  \item \takeaways{Our question:} can a \textbf{simple} approximation capture the mapping \emph{as a whole}? 
\end{itemize}

\postersec{1.\ The Mapping has high predictive Power and is mostly Linear}
\begin{center}
\includegraphics[width=0.95\linewidth]{plot1_scaling_boundary.pdf}
\end{center}
\begin{itemize}
  \item \takeaways{The mapping has high predictive power:} held-out $R^2$ 0.75, rank-1 retrieval 0.98 at 1 million contexts
  \item \takeaways{The mapping is mostly linear:} the nonlinear (MLP) map adds only $\sim$$0.06$ $R^2$ over the linear map, and only at large $n$ (potentially this advantage would grow as you scale up the number of contexts)
  \item \takeaways{The mapping is not present for generic boundary tokens and is not just copying the context vector (baseline)}
  \item We study the \textbf{linear} map here and leave more complex maps to future work
\end{itemize}

\postersec{2.\ The mapping predicts useful parts of the answer}
\begin{center}
\IfFileExists{figures/plot2_persona_directions.pdf}{\includegraphics[width=0.86\linewidth]{plot2_persona_directions.pdf}}{\includegraphics[width=0.46\linewidth]{c3_persona_direction_spectrum.pdf}}
\end{center}
\begin{itemize}
  \item \takeaways{Persona-vector directions are among the best-predicted directions of the answer:} $R^2$ \textbf{0.80--0.91} against a random-direction mean of \textbf{0.67}, and every one does better than 95\% of random directions (0.76)
\end{itemize}

\postersec{3.\ The mapping is better at predicting high-level and identity related features of the answer}
We train another map from the context to the answer's \textbf{SAE features}, then partial out predictors \emph{one at a time} to see which survives.
\begin{center}
\includegraphics[width=0.90\linewidth]{plot3_sae_predictors.pdf}
\end{center}
\begin{itemize}
  \item Controlling for mean activation, the best predictors are
  \begin{itemize}
      \item being related to speaker identity
      \item not suppressing/promoting a few tokens
      \item not being related to topic
      \item being in the coarsest tier of a Matryoshka SAE (intuitively, being high-level)
  \end{itemize}
  \item Therefore, the mapping seems to be 
  \begin{itemize}
      \item good at predicting \textbf{high-level} and \textbf{identity-related} features
      \item bad at predicting \textbf{low-level} and \textbf{topic-related} features
      \item suggests it might be some kind of \textbf{persona mapping}
  \end{itemize} 
\end{itemize}

\postersec{5.\ The mapping is already largely present in the base model and refined throughout post training}
\begin{center}
\IfFileExists{figures/plot5_stages.pdf}{\includegraphics[width=0.90\linewidth]{plot5_stages.pdf}}{\includegraphics[width=0.38\linewidth]{c1_stage_retention.pdf}}
\end{center}
\begin{itemize}
  \item \takeaways{The mapping is already largely present in the base model:} $R^2$ 0.41 before any post-training, \textbf{67\%} of the 0.62 the fully post-trained model reaches.
  \item \takeaways{SFT has the largest effect on the mapping} ($+0.17$), with DPO and both RLVR stages together adding only $+0.04$.
\end{itemize}

\postersec{6.\ The mapping is shared with story characters}
\begin{center}
\includegraphics[width=0.76\linewidth]{plot6c_pooled_lattice.pdf}
\end{center}
\begin{itemize}
  \item \takeaways{One shared map does almost as well as per-character/framing maps:} 62-94\% of each framing/character's own-map performance
  \begin{itemize}
      \item This suggests the model is representing the assistant in the chat template similarly to a character in a story, providing evidence for the \textbf{Persona Selection Model}
  \end{itemize}

    \item \takeaways{Characters more similar to the assistant have better mappings}
    \begin{itemize}
        \item But this difference only arises in the instruct model: the assistant's privileged position/representation as a persona comes from \textbf{post-training}
    \end{itemize}
\end{itemize}

\postersec{7. Application: behavior prediction before inference}
\begin{center}
\IfFileExists{figures/plot9_application_arms.pdf}{\includegraphics[width=0.99\linewidth]{plot9_application_arms.pdf}}{\includegraphics[width=0.99\linewidth]{c5_pv_methods_regimes.pdf}}
\end{center}
\begin{itemize}
  \item \takeaways{Persona vectors projected on the context vector fail to generalize to realistic contexts}
  \begin{itemize}
      \item Projecting persona vectors on the mapped answer does almost as well as projecting on the real answer
  \end{itemize}
  \item \takeaways{Regression on the mapped answer does the best of all methods} 
  \begin{itemize}
      \item This showcases one potential application of this mapping
  \end{itemize}
\end{itemize}

\postersec{Other Results}
\begin{itemize}
  \item \takeaways{The mapping does not get reliably better with model scale}
  \item \takeaways{In thinking models, the mapping predicts just as well from the pre-CoT context vector and the post-CoT context vector}
  \begin{itemize}
      \item suggesting that whatever is predicted by the mapping is not affected by CoT
  \end{itemize}
\end{itemize}

\postersec{Implications}
\begin{itemize}
    \item A  linear mapping can approximate much of the context-answer mapping implemented by a LLM
    \item Applications: predicting behavior, automated redteaming, predicting effect of fine-tuning
    \item Evidence for persona selection model
    \item Motivates further work into metamodels of answer activations and the linearity in transformers
\end{itemize}

\postersec{Future work (mostly in progress)}
\begin{itemize}
  \item Trying more complex estimators (diffusion, flow matching, etc.)
  \item Predicting the entire answer distribution to bound probability of rare behaviors 
  \item Theoretical analysis of the mapping 
  \item Predicting other properties of the answer (correctness, refusal, etc.)
  \item Predicting the effect of fine-tuning on the mapping/model
  \item Automated redteaming / jailbreaking.
  \item Single-token next-activation mapping
\end{itemize}
