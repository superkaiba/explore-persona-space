% --------------------------------------------------------------------------
% Panel A: explicit token-by-layer representation view.

\node[title] at (0.0,4.25) {(a) Representations at transformer layer $\ell$};

% Token row.
\node[token] (c1) at (1.15,0.05) {User:};
\node[token] (c2) at (2.05,0.05) {$\cdots$};
\node[token, minimum width=12mm] (ct) at (3.15,0.05) {last context};
\node[token, draw=ans!65, fill=ans!6] (a1) at (4.45,0.05) {$a_1$};
\node[token, draw=ans!65, fill=ans!6] (a2) at (5.35,0.05) {$\cdots$};
\node[token, draw=ans!65, fill=ans!6] (am) at (6.25,0.05) {$a_{T_A}$};

  ($(a1.south west)+(0,-1mm)$) -- ($(am.south east)+(0,-1mm)$)

  \draw[->, thin, black!25] (\x,2.46) -- (\x,3.13);
}

% Highlighted layer-l states.
\node[state] at (1.15,2.05) {};
