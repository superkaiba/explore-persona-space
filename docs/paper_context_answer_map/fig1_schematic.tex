% Figure 1: context-to-answer prediction pipeline (rev 3, 2026-09-03).
% Compile standalone: pdflatex fig1_schematic.tex
%
% Design notes (Thomas, 2026-09-03): no circles for vectors (cell strips instead),
% Inter text font to match the other paper figures, a concrete example rendered
% exactly as Qwen2.5-7B-Instruct's chat template renders it (the context vector
% sits on the "\n" token after "assistant", token id 198), the ridge predictor
% drawn as a matrix plus a bias with the MLP as a small alternative, and no
% evaluation / uses boxes. Palette = paper palette (c2a_plot_style.py):
% ink #22272B, muted #687078, linear teal #176B87, nonlinear terracotta #C4553D.
% Two text sizes only: 8 pt (symbols, names) and 7 pt (annotations, transcript).
\documentclass[tikz,border=4pt]{standalone}
\usepackage[type1]{inter}
\renewcommand{\familydefault}{\sfdefault}
\usepackage[scaled=0.96]{inconsolata}
\usepackage{amsmath}
\usetikzlibrary{positioning,arrows.meta,calc,fit}

\definecolor{ink}{HTML}{22272B}
\definecolor{muted}{HTML}{687078}
\definecolor{teal}{HTML}{176B87}
\definecolor{terra}{HTML}{C4553D}
\definecolor{paper}{HTML}{FFFFFF}

% ---- vector glyph: a row of small square cells ------------------------------
% \vecstrip{<node name>}{<x>}{<y>}{<cell color>}
% 7 cells, 2.0 mm each, 0.35 mm gap; fixed pseudo-random intensities.
\newcommand{\vecstrip}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \fill[paper] (-0.84,-0.115) rectangle (0.84,0.115);
    \foreach \i/\s in {0/38,1/14,2/56,3/24,4/66,5/18,6/44} {
      \fill[#4!\s] ({-0.8225+\i*0.235},-0.1) rectangle ++(0.2,0.2);
    }
    \draw[#4!70, line width=0.45pt, rounded corners=0.4pt] (-0.84,-0.115) rectangle (0.84,0.115);
  \end{scope}
  \node[inner sep=0pt, minimum width=1.68cm, minimum height=0.23cm] (#1) at (#2,#3) {};
}
% ---- matrix glyph: a 6x6 grid --------------------------------------------------
% \matglyph{<node name>}{<x>}{<y>}{<color>}   cell 1.35 mm, gap 0.25 mm
\newcommand{\matglyph}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \r/\row in {0/{40,18,62,26,50,14},1/{16,54,22,44,30,60},2/{58,28,36,12,64,34},
                        3/{24,46,14,56,20,48},4/{50,12,60,32,42,26},5/{30,62,26,48,16,54}} {
      \foreach \c/\s [count=\ci from 0] in \row {
        \fill[#4!\s] ({-0.48+\ci*0.16},{0.32-\r*0.16}) rectangle ++(0.135,0.135);
      }
    }
    \draw[#4!75, line width=0.45pt, rounded corners=0.4pt] (-0.495,-0.495) rectangle (0.495,0.495);
  \end{scope}
  \node[inner sep=0pt, minimum width=0.99cm, minimum height=0.99cm] (#1) at (#2,#3) {};
}
% ---- column glyph: 6 stacked cells -----------------------------------------------
\newcommand{\colglyph}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \r/\s in {0/44,1/20,2/58,3/30,4/16,5/52} {
      \fill[#4!\s] (-0.0675,{0.32-\r*0.16}) rectangle ++(0.135,0.135);
    }
    \draw[#4!75, line width=0.45pt, rounded corners=0.4pt] (-0.0825,-0.495) rectangle (0.0825,0.495);
  \end{scope}
  \node[inner sep=0pt, minimum width=0.165cm, minimum height=0.99cm] (#1) at (#2,#3) {};
}

% ===========================================================================

\node[lbl, anchor=west, font=\fontsize{8}{9.5}\selectfont\bfseries] at (-0.12,5.08)
  {Example context and sampled answer};

% transcript backdrop
\draw[fill=black!3, draw=black!22, rounded corners=2pt] (-0.12,2.42) rectangle (5.22,4.88);

\node[monomuted] (l1) at (0.0,4.66) {<|im\_start|>system};
\node[monomuted] (l2) at (0.0,4.34) {You are Qwen, created by Alibaba Cloud.};
\node[monomuted] (l3) at (0.32,4.02) {You are a helpful assistant.<|im\_end|>};  % soft wrap of line 2 (hanging indent)
\node[monomuted] (l4) at (0.0,3.70) {<|im\_start|>user};
\node[mono]      (l5) at (0.0,3.38) {What is the capital of France?\textcolor{muted}{<|im\_end|>}};
\node[monomuted] (l6) at (0.0,3.06) {<|im\_start|>assistant};
% the "\n" token that ends the context (Qwen token id 198)
\node[mono, draw=ink, line width=0.6pt, fill=paper, rounded corners=1pt, inner sep=1.3pt,

% the sampled answer, then the end-of-turn token
\node[mono, fill=black!11, rounded corners=1pt, inner sep=1.3pt, anchor=west] (ans) at (0.0,2.68)
  {The capital of France is Paris.};
\node[monomuted, anchor=west] (eot) at ($(ans.east)+(0.05,0)$) {<|im\_end|>};

% brackets: context C spans lines 1-6 incl. "\n"; answer A_k is line 7
\draw[ink!55, line width=0.5pt] (-0.30,4.80) -- (-0.38,4.80) -- (-0.38,2.94) -- (-0.30,2.94);
\node[ann, rotate=90, anchor=south, text=ink] at (-0.42,3.87) {context $C$};
\draw[ink!55, line width=0.5pt] (-0.30,2.84) -- (-0.38,2.84) -- (-0.38,2.52) -- (-0.30,2.52);
\node[ann, rotate=90, anchor=south, text=ink] at (-0.42,2.68) {$A_k$};

% ===========================================================================
% Middle: the two representations at layer l.
% ===========================================================================
% context vector (top row)
\vecstrip{hc}{7.95}{4.20}{ink}
\node[lbl, above=1.5pt of hc] {context vector $h_C^{\ell}$};
\draw[flow] (nl.east) -- (5.42,3.06) -- (5.42,4.20) -- (hc.west);
\node[ann, anchor=north] at (6.48,4.08) {layer-$\ell$ state at the\\final context token};

% answer vector (bottom row)
\vecstrip{ha}{7.95}{1.35}{ink}

\draw[flow] (eot.east) -- (5.58,2.68) -- (5.58,1.35) -- (ha.west);
\node[ann, anchor=south] at (6.62,1.50) {mean of the layer-$\ell$\\states over $A_k$'s tokens};

% ===========================================================================
% Right: the predictor (top) and the prediction target (bottom).
% ===========================================================================
% ---- ridge predictor box
\node[draw=teal, line width=0.7pt, fill=teal!5, rounded corners=2.5pt,

\node[draw=terra, dashed, line width=0.55pt, fill=paper, rounded corners=2pt,

% ---- prediction target: mean over K sampled answers

\vecstrip{s2}{10.53}{1.44}{ink}

\vecstrip{mu}{13.20}{1.35}{ink}
\node[lbl, below=1.5pt of mu] {target $\bar\mu_A^{\ell}$};

\node[ann, anchor=south] at (12.18,1.50) {mean over\\$K$ rollouts};

% ---- fit: prediction against target

\node[ann, text=teal, fill=paper] at (13.20,2.95) {ridge fit:\\$\min\,\lVert \bar\mu_A^{\ell}-\hat h_A^{\ell}\rVert^{2}$};
