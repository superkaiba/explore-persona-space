% Figure 1: context-to-answer prediction pipeline (rev 6, 2026-09-03).
% Compile standalone: pdflatex fig1_schematic.tex
% THEN convert text to outlines before copying to Overleaf (REQUIRED):
%   gs -q -dNOPAUSE -dBATCH -sDEVICE=pdfwrite -dNoOutputFonts -dCompatibilityLevel=1.5 \
%      -sOutputFile=fig1_schematic_outlined.pdf fig1_schematic.pdf
% Why: the Computer Modern macron (\bar), minus and \cdots glyphs sit at character
% codes 22, 0 and 1 of Type 1 fonts with built-in encodings. pdf.js and Ghostscript
% resolve them, but the Chrome/Edge (Adobe-powered) PDF engines fail the lookup and
% draw a literal "No Glyph" box over every bar accent (Thomas, 2026-09-03). Outlining
% removes all fonts from the figure PDF, so no viewer does any glyph lookup.
%
% Design (Thomas, 2026-09-03): follows the residual-stream convention of Hendel et al.
% 2023 (task vectors, Fig. 1): a model box holding a grid of tall rounded hidden-state
% rectangles, tokens as columns, layers as rows with dots for the elided layers, the
% extracted states filled in a solid color. Vectors are tall contiguous-cell columns
% (no gaps), the ridge map is a contiguous 6x6 cell grid, so the predictor reads as
% matrix x column + column. Tokens are truncated but the template tokens around the
% context/answer boundary are exact for Qwen2.5-7B-Instruct (the context vector sits
% on the "\n" token after "assistant", token id 198). Palette = paper palette
% (c2a_plot_style.py): ink #22272B, muted #687078, linear teal #176B87, nonlinear
% terracotta #C4553D. Two text sizes: 8 pt (names) and 7 pt (annotations, tokens).

% ---- column vector glyph: 6 contiguous cells, 0.34 x 1.20 cm ----------------------
% \colvec{<node name>}{<x>}{<y>}{<color>}
\newcommand{\colvec}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \r/\s in {0/42,1/16,2/60,3/26,4/68,5/20} {
      \fill[#4!\s] (-0.17,{0.60-(\r+1)*0.20}) rectangle (0.17,{0.60-\r*0.20});
    }
    \draw[#4!75, line width=0.5pt] (-0.17,-0.60) rectangle (0.17,0.60);
  \end{scope}
  \node[inner sep=0pt, minimum width=0.34cm, minimum height=1.2cm] (#1) at (#2,#3) {};
}
% ---- matrix glyph: 6x6 contiguous cells, 1.20 x 1.20 cm ---------------------------
\newcommand{\matgrid}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \r/\row in {0/{40,18,62,26,50,14},1/{16,54,22,44,30,60},2/{58,28,36,12,64,34},
                        3/{24,46,14,56,20,48},4/{50,12,60,32,42,26},5/{30,62,26,48,16,54}} {
      \foreach \s [count=\ci from 0] in \row {
        \fill[#4!\s] ({-0.60+\ci*0.20},{0.60-(\r+1)*0.20}) rectangle ({-0.60+(\ci+1)*0.20},{0.60-\r*0.20});
      }
    }
    \draw[#4!75, line width=0.5pt] (-0.60,-0.60) rectangle (0.60,0.60);
  \end{scope}
  \node[inner sep=0pt, minimum width=1.2cm, minimum height=1.2cm] (#1) at (#2,#3) {};
}
