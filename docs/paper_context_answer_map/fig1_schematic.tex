% Figure 1: context-to-answer prediction pipeline (rev 13, 2026-09-03).
% Compile standalone: pdflatex fig1_schematic.tex
% THEN convert text to outlines before copying to Overleaf (REQUIRED):
%   gs -q -dNOPAUSE -dBATCH -sDEVICE=pdfwrite -dNoOutputFonts -dCompatibilityLevel=1.5 \
%      -sOutputFile=fig1_schematic_outlined.pdf fig1_schematic.pdf
% Why: the Computer Modern macron (\bar), minus and \cdots glyphs sit at character
% codes 22, 0 and 1 of Type 1 fonts with built-in encodings. pdf.js and Ghostscript
% resolve them, but the Chrome/Edge (Adobe-powered) PDF engines fail the lookup and
% draw a literal "No Glyph" box over every bar accent (Thomas, 2026-09-03). Outlining
% removes all fonts from the figure PDF, so no viewer does any glyph lookup.
%
% Design (Thomas, 2026-09-03): the model is a grid of per-token residual streams
% (Hendel et al. 2023 convention). Each token column is ONE continuous vertical line,
% the residual stream, with the layer blocks and the residual-stream states (5-cell
% columns) sitting on it; dots elide layers. The row of states after layer l is the
% shaded band, labeled "residual stream after layer l" on the left. The final context
% token's state (violet) is tapped upward through the gap between the context and
% answer columns and leaves the model through the strip above layer L; the answer
% tokens' states (amber, grouped) are averaged into the answer vector. Every vector is
% a contiguous 5-cell column, the ridge map a contiguous 5x5 grid, so the predictor
% reads as matrix x column + column. Tokens are truncated but the template tokens
% around the context/answer boundary are exact for Qwen2.5-7B-Instruct (the context
% vector sits on the "\n" token after "assistant", token id 198).
% Palette = paper palette (c2a_plot_style.py): ink #22272B, muted #687078, linear teal
% #176B87. Two text sizes: 8 pt (names) and 7 pt.
% Side coding (rev 13, Thomas): every context-side state and vector is violet #6F4FA0
% (all context tokens' states after layer l, h_C, the predictor input). Answer side is an
% amber ramp that darkens with aggregation: token states light amber #C98A1B, the mean
% answer vectors h_A(A_k) / h_A(A_1..K) darker amber #A8720F, the target (mean over K)
% darkest #6E4A0C. Vectors outside the model use stronger cell shades than the states.
% The prediction stays teal, the predictor's output.
% Rev 12 layout (Thomas: "raise it even more"): the model box top is level with the
% predictor box top; the context tap runs in a 0.6 cm strip above layer L; the top-row
% vector labels sit BELOW their vectors (prediction label to the right) so nothing
% rises above the two box titles; the answer row steps 0.9 cm below the band so the
% right-hand column ends near the model box bottom.

% ---- big column vector: 5 contiguous SQUARE cells, 0.20 x 1.00 cm -------------------
% \colvecs{<node name>}{<x>}{<y>}{<color>}{<shade list>}   \colvec = default shades
\newcommand{\colvecs}[5]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \s [count=\r from 0] in {#5} {
      \fill[#4!\s] (-0.10,{0.50-(\r+1)*0.20}) rectangle (0.10,{0.50-\r*0.20});
    }
    \draw[#4!75, line width=0.5pt] (-0.10,-0.50) rectangle (0.10,0.50);
  \end{scope}
  \node[inner sep=0pt, minimum width=0.20cm, minimum height=1.0cm] (#1) at (#2,#3) {};
}
\newcommand{\colvec}[4]{\colvecs{#1}{#2}{#3}{#4}{42,16,60,26,68}}
% ---- small residual-stream state inside the model: 5 SQUARE cells, 0.10 x 0.50 cm ---
% \minivec{<x>}{<y>}{<color>}{<shade list>}{<outline>}
\newcommand{\minivec}[5]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \s [count=\r from 0] in {#4} {
      \fill[#3!\s] (-0.05,{0.25-(\r+1)*0.10}) rectangle (0.05,{0.25-\r*0.10});
    }
    \draw[#5, line width=0.4pt] (-0.05,-0.25) rectangle (0.05,0.25);
  \end{scope}
}
% ---- matrix glyph: 5x5 contiguous cells, 1.00 x 1.00 cm ---------------------------
\newcommand{\matgrid}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \r/\row in {0/{40,18,62,26,50},1/{16,54,22,44,30},2/{58,28,36,12,64},
                        3/{24,46,14,56,20},4/{50,12,60,32,42}} {
      \foreach \s [count=\ci from 0] in \row {
        \fill[#4!\s] ({-0.50+\ci*0.20},{0.50-(\r+1)*0.20}) rectangle ({-0.50+(\ci+1)*0.20},{0.50-\r*0.20});
      }
    }
    \draw[#4!75, line width=0.5pt] (-0.50,-0.50) rectangle (0.50,0.50);
  \end{scope}
  \node[inner sep=0pt, minimum width=1.0cm, minimum height=1.0cm] (#1) at (#2,#3) {};
}
% ---- one residual stream per token column -----------------------------------------
% \stream{<x>}{<token label style>}{<token>}{<layer-l state color>}{<shades>}{<outline>}
% One continuous vertical line (the residual stream) from the token up to layer L; the
% layer blocks and the states sit ON the line, drawn after it; dots elide layers.
% Rows: token 1.54, block 1 at 2.07, state 2.62, dots 3.00-3.18, block l 3.55,
%       state 4.10 (the read row), dots 4.48-4.66, block L 5.03.
\newcommand{\stream}[6]{%
  \node[#2] at (#1,1.54) {#3};
  \draw[mini] (#1,1.73) -- (#1,1.94);
  \draw[streamline] (#1,2.19) -- (#1,2.93);
  \fill[black!45] (#1,3.00) circle (0.5pt) (#1,3.09) circle (0.5pt) (#1,3.18) circle (0.5pt);
  \draw[mini] (#1,3.25) -- (#1,3.42);
  \draw[streamline] (#1,3.67) -- (#1,4.41);
  \fill[black!45] (#1,4.48) circle (0.5pt) (#1,4.57) circle (0.5pt) (#1,4.66) circle (0.5pt);
  \draw[mini] (#1,4.73) -- (#1,4.90);
  \node[blk] at (#1,2.07) {};
  \node[blk] at (#1,3.55) {};
  \node[blk] at (#1,5.03) {};
  \minivec{#1}{2.62}{muted}{40,18,54,26,62}{muted!70}
  \minivec{#1}{4.10}{#4}{#5}{#6}
}
