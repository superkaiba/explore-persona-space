% Figure 1: context-to-answer prediction pipeline (rev 4, 2026-09-03).
% Compile standalone: pdflatex fig1_schematic.tex
%
% Design notes (Thomas, 2026-09-03): show the transformer layers (token columns
% x layer rows, layer l highlighted) so the two vectors are read off the layer-l
% row; tokens may be truncated ("..." columns) but the template tokens around the
% context/answer boundary are exact for Qwen2.5-7B-Instruct (the context vector
% sits on the "\n" token after "assistant", token id 198). No circles for vectors
% (cell strips), Inter text font matching the other paper figures, the ridge
% predictor drawn as a matrix plus a bias with the MLP as a small dashed
% alternative, no evaluation / uses boxes. Palette = paper palette
% (c2a_plot_style.py): ink #22272B, muted #687078, linear teal #176B87,
% nonlinear terracotta #C4553D. Two text sizes: 8 pt (names) and 7 pt (annotations, tokens).

% ---- vector glyph: a row of small square cells ------------------------------
\newcommand{\vecstrip}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \fill[paper] (-0.84,-0.115) rectangle (0.84,0.115);
    \foreach \i/\s in {0/38,1/14,2/56,3/24,4/66,5/18,6/44} {
      \fill[#4!\s] ({-0.8225+\i*0.235},-0.1) rectangle ++(0.2,0.2);
    }
    \draw[#4!70, line width=0.45pt, rounded corners=0.4pt] (-0.84,-0.115) rectangle (0.84,0.115);
  \end{scope}
  \node[inner sep=0pt, minimum width=1.68cm, minimum height=0.23cm] (#1) at (#2,#3) {};
}
% ---- matrix glyph: a 6x6 grid --------------------------------------------------
\newcommand{\matglyph}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \r/\row in {0/{40,18,62,26,50,14},1/{16,54,22,44,30,60},2/{58,28,36,12,64,34},
                        3/{24,46,14,56,20,48},4/{50,12,60,32,42,26},5/{30,62,26,48,16,54}} {
      \foreach \c/\s [count=\ci from 0] in \row {
        \fill[#4!\s] ({-0.48+\ci*0.16},{0.32-\r*0.16}) rectangle ++(0.135,0.135);
      }
    }
    \draw[#4!75, line width=0.45pt, rounded corners=0.4pt] (-0.495,-0.495) rectangle (0.495,0.495);
  \end{scope}
  \node[inner sep=0pt, minimum width=0.99cm, minimum height=0.99cm] (#1) at (#2,#3) {};
}
% ---- column glyph: 6 stacked cells -----------------------------------------------
\newcommand{\colglyph}[4]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \r/\s in {0/44,1/20,2/58,3/30,4/16,5/52} {
      \fill[#4!\s] (-0.0675,{0.32-\r*0.16}) rectangle ++(0.135,0.135);
    }
    \draw[#4!75, line width=0.45pt, rounded corners=0.4pt] (-0.0825,-0.495) rectangle (0.0825,0.495);
  \end{scope}
  \node[inner sep=0pt, minimum width=0.165cm, minimum height=0.99cm] (#1) at (#2,#3) {};
}

% ===========================================================================
% Left: transformer residual stream, tokens as columns, layers as rows.
% Context columns i = 0..8 at x = 0.75 + 0.42 i; answer columns j = 0..4 at x = 4.75 + 0.42 j.

  {Residual stream};

% layer labels
\node[ann, anchor=east, text=ink] at (0.45,2.30) {layer $L$};
\node[ann, anchor=east] at (0.45,1.96) {$\vdots$};
\node[ann, anchor=east, text=ink] at (0.45,1.62) {layer $\ell$};
\node[ann, anchor=east] at (0.45,1.285) {$\vdots$};
\node[ann, anchor=east, text=ink] at (0.45,0.95) {layer $1$};

% layer-l band
