**Grain-claim recognizer (the regexes, drafted):**

```python
# The per-point unit head noun must be draw/row, reached from the point
# token through <=2 letter-leading modifier tokens ("one steered prefill
# draw"). A digit token, a non-letter symbol (×), or punctuation breaks
# the bridge — "one pair at greedy 1 draw" (#2162), "one cell × estimator
# row" (#602), and "one of the 432 pooled rows" (#504) never match.
# (?!['’-]) blocks possessives + hyphen-modifier usage ("one probe row's
# paired margin delta" #1489, "draw-averaged").
_GRAIN_MOD = r"(?:[a-z][\w-]{0,24}\s+){0,2}?"
_GRAIN_EACH_RE = re.compile(
    r"\b(?:each|every)\s+(?:point|dot|marker)s?\s+"
    r"(?:is|=|represents|shows|corresponds\s+to)\s+"
    r"(?:one|a|an)\s+" + _GRAIN_MOD + r"(?P<unit>draw|row)s?\b(?!['’-])",
    re.IGNORECASE,
)
_GRAIN_PER_RE = re.compile(
    r"\bone\s+(?:point|dot|marker)\s+per\s+" + _GRAIN_MOD
    + r"(?P<unit>draw|row)\b(?!['’-])",
    re.IGNORECASE,
)
_GRAIN_ADJ_RE = re.compile(
    r"\bper[- ](?P<unit>draw|row)\s+(?:point|dot|marker|scatter)s?\b",
    re.IGNORECASE,
)
```

All quantifiers are bounded (`{0,2}` lazy over `{0,24}`-bounded token atoms — the check-45 `_CC_GAP` bounded-gap posture; no nested unbounded repetition, so no catastrophic backtracking on ~50 KB bodies).

**Body-side declaration harvest (once per body, pure regex, before any git read):**

```python
# K (decode draws per cell). Deliberately narrow: only the two strongly
# decode-flavored forms; "N=5 draws" (judge draws, llm-judging rule) and
# hyphenated bootstrap forms ("10,000-draw bootstrap") never match.
_GRAIN_K_RES = (
    re.compile(r"\bK\s*=\s*(\d{1,2})\s*draws?\b", re.IGNORECASE),
    re.compile(r"(?<![A-Za-z0-9])[×x]\s*(\d{1,2})\s+draws?\b", re.IGNORECASE),
)
# Reject a K match whose preceding 40 chars carry resampling vocabulary:
_GRAIN_K_GUARD_RE = re.compile(r"bootstrap|permutation|resampl|null[- ]draw", re.IGNORECASE)
# Accepted K bounds: 2 <= K <= 64 (K=1 makes draw grain == cell grain).

# Pair-flavored counts only (NOT bare n= anywhere — see §5 false-positive
# analysis; word-bounded so "paired" never matches):
_GRAIN_PAIRCOUNT_RE = re.compile(
    r"\b(\d{1,4})\s+(?:[a-z][\w-]{0,24}\s+){0,3}?(?:pairs|survivors)\b", re.IGNORECASE
)
_GRAIN_NEQ_RE = re.compile(r"\bn\s*=\s*(\d{1,4})\b", re.IGNORECASE)
_GRAIN_PAIRISH_RE = re.compile(r"\bpairs?\b|\bsurviv", re.IGNORECASE)
# An n=NNN token is harvested ONLY when _GRAIN_PAIRISH_RE hits within
# ±60 chars of the match (word-bounded — "paired" does not arm it).
# Accepted pair-count bounds: 2 <= P <= 5000; keep at most the first 24
# distinct values (bound the product set).
```

**Candidate products (the "≈ pairs × schemes" arithmetic, exact spec of "≈"):**

```python
def _grain_products(pair_counts, ks):
    cands = {p: str(p) for p in pair_counts}
    for a, b in itertools.combinations(sorted(pair_counts), 2):
        cands.setdefault(a + b, f"{a}+{b}")       # pooled pair SETS (the incident
                                                   # figure pools 180 fmt + 15 mq = 195)
    agg = {}                                       # product -> (repr, s) for the WARN text
    for p, rep in cands.items():
        for s in (1, 2, 3, 4):                     # scheme/arm multiplier; parsing the
            agg.setdefault(p * s, (rep, s))        # donor-scheme count from prose is
    draw = {m * k for m in agg for k in ks}        # fragile — a small-integer set is not
    return agg, draw

def _grain_near(n, m):                             # "≈" := within max(2, 2% of m)
    return abs(n - m) <= max(2, round(0.02 * m))   # 2 absorbs the incident's judge-drop
                                                   # attrition (388 vs 390); 2% scales it
```

**Per-series counts from the sidecar:**

```python
def _sidecar_series_point_counts(meta):
    if meta.get("data_truncated") or meta.get("truncated"):
        return None                                # counts over truncated rows are not
    pts = meta.get("points")                       # figure truth (check-45 parity)
    if not isinstance(pts, list):
        pts = meta.get("rows")                     # legacy sidecars
    if not isinstance(pts, list):
        return None
    rows = [p for p in pts if isinstance(p, dict)]
    kinds = {p.get("_kind") for p in rows if isinstance(p.get("_kind"), str)}
    if "scatter" in kinds:                         # grain claims are about POINTS;
        rows = [p for p in rows if p.get("_kind") == "scatter"]   # drop bar/line rows
    counts: dict[str, int] = {}
    for p in rows:
        key = p["series"] if isinstance(p.get("series"), str) else str(p.get("_group", ""))
        counts[key] = counts.get(key, 0) + 1
    return counts or None
```
