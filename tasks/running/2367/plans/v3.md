**Grain-claim recognizer (the regexes, drafted; verified by execution at plan-v3 time — §5):**

```python
# The per-point unit head noun must be draw/row, reached from the point
# token through <=3 letter-leading modifier tokens ("one steered prefill
# draw"; "one single steered prefill draw" — the founding caption carries
# exactly 2 modifiers, so {0,2} had ZERO headroom on the founding class;
# widened to {0,3} per MF-4, measured to add zero new corpus hit files).
# A digit token, a non-letter symbol (×), or punctuation breaks the
# bridge — "one pair at greedy 1 draw" (#2162; the DIGIT token is what
# rejects it at {0,3} — measured), "one cell × estimator row" (#602), and
# "one of the 432 pooled rows" (#504) never match. (?!['’-]) blocks
# possessives + hyphen-modifier usage ("one probe row's paired margin
# delta" #1489, "draw-averaged").
_GRAIN_MOD = r"(?:[a-z][\w-]{0,24}\s+){0,3}?"
_GRAIN_EACH_RE = re.compile(
    r"\b(?:each|every)\s+(?:point|dot|marker)s?\s+"
    r"(?:is|=|represents|shows|corresponds\s+to)\s+"
    r"(?:one|a|an)\s+" + _GRAIN_MOD + r"(?P<unit>draw|row)s?\b(?!['’-])",
    re.IGNORECASE,
)
_GRAIN_PER_RE = re.compile(
    r"\bone\s+(?:point|dot|marker)\s+per\s+" + _GRAIN_MOD
    + r"(?P<unit>draw|row)\b(?!['’-])",
    re.IGNORECASE,
)
# Adjective form: 'marker' is accepted as the glyph for per-DRAW only.
# "per-row marker …" is the ※-marker DOMAIN TERM in this repo (#399
# "per-row marker-loss contribution", #471 "Per-row marker count
# (`text.count(\"※\")`)"): the trailing-noun vocabulary is open-ended
# (count, loss, token count, Δlogp, …) and #399's HYPHENATED form
# ("marker-loss") defeats any `\s+`-lookahead noun guard — measured at
# plan time — so the row+marker combination is excluded outright.
# Named recall sacrifice: a genuine per-row glyph caption saying
# "per-row markers" is missed (glyph captions in this corpus use
# point/dot); the EACH/PER copular forms above still accept marker
# with unit=row.
_GRAIN_ADJ_RE = re.compile(
    r"\bper[- ](?:(?P<unit_d>draw)\s+(?:point|dot|marker|scatter)s?"
    r"|(?P<unit_r>row)\s+(?:point|dot|scatter)s?)\b",
    re.IGNORECASE,
)   # unit = "draw" if m["unit_d"] else "row"
```

**Body-side declaration harvest (once per body, pure regex, before any git read):**

```python
# K (decode draws per cell). Deliberately narrow: only the two strongly
# decode-flavored forms; "N=5 draws" (judge draws, llm-judging rule) and
# hyphenated bootstrap forms ("10,000-draw bootstrap") never match.
# Named disarm sacrifices (K-vocabulary classes that never arm the check):
# "10 on-policy rollouts per question", "5 samples per prompt", table-cell
# "| Draws per cell | 5 |", spelled-out "five draws" — all disarm at
# ladder step 3 (doc bullet).
_GRAIN_K_RES = (
    re.compile(r"\bK\s*=\s*(\d{1,2})\s*draws?\b", re.IGNORECASE),
    re.compile(r"(?<![A-Za-z0-9])[×x]\s*(\d{1,2})\s+draws?\b", re.IGNORECASE),
)
# Reject a K match whose preceding 40 chars carry resampling vocabulary:
_GRAIN_K_GUARD_RE = re.compile(r"bootstrap|permutation|resampl|null[- ]draw", re.IGNORECASE)
# Accepted K bounds: 2 <= K <= 64 (K=1 makes draw grain == cell grain).

# Pair-flavored counts only (NOT bare n= anywhere — see §5 false-positive
# analysis; word-bounded so "paired" never matches). Named sacrifice:
# non-pair-unit designs (question/context/persona-grain declarations)
# never arm the pair harvest — the check is pair-unit-scoped by design.
_GRAIN_PAIRCOUNT_RE = re.compile(
    r"\b(\d{1,4})\s+(?:[a-z][\w-]{0,24}\s+){0,3}?(?:pairs|survivors)\b", re.IGNORECASE
)
_GRAIN_NEQ_RE = re.compile(r"\bn\s*=\s*(\d{1,4})\b", re.IGNORECASE)
_GRAIN_PAIRISH_RE = re.compile(r"\bpairs?\b|\bsurviv", re.IGNORECASE)
# An n=NNN token is harvested ONLY when _GRAIN_PAIRISH_RE hits within
# ±60 chars of the match (word-bounded — "paired" does not arm it).
# Accepted pair-count bounds: 2 <= P <= 5000; keep at most the first 24
# distinct values (bound the product set). SHOULD: when the 24-cap is
# hit, the WARN detail appends "(pair-count harvest capped at 24 distinct
# values)" — a late true-count veto can be dropped by the document-order
# cap (Concerns handed to the implementer, item 1).

# RANGE-ENDPOINT GUARD (the v1→v2 blocker fix, EXTENDED to worded
# `to`/`through` ranges per MF-1; applied to BOTH pair regexes'
# captured-digit spans). A number that is an endpoint of a numeric range —
# dash-spelled ("n=26–36 pairs per cell" — #2333 pre-fix line 200;
# "n=148–150 pairs after drops" — #1489) or WORD-spelled ("N = 50 to 550
# pairs" — #207:68, corpus-attested; "26 through 36 pairs") — is a
# per-cell / after-drops RANGE, not a plotted-total candidate. v1
# harvested 26 from the real pre-fix #2333 body and the spurious draw
# product 26×3×5=390 vetoed the incident WARN (|388−390|=2 ≤ tol 8).
# Excluding endpoints prunes BOTH product sets (agg and draw both derive
# from the pair harvest): it removes spurious agg (fire-side) AND spurious
# draw (veto-side) candidates — on the incident it prunes the spurious
# veto and RESTORES the fire; the symmetric residual (a range-only true
# count losing its veto contribution) is residual (i) in §5. A digit is
# required on the far side of the dash/word, so hyphenated compounds
# ("2-token", "1-token prefill") and prepositional "compared to 36 pairs" /
# "scaled up to 36 pairs" are never rejected. En dash, em dash, minus,
# ASCII hyphen, and the words to/through all count. Context windows are
# ±16 chars (worded forms are longer: " through 3" alone is 10 chars).
_GRAIN_RANGE_TAIL_RE = re.compile(
    r"\s*[–—−-]\s*\d|\s+(?:to|through)\s+\d", re.IGNORECASE)   # "26" in "26–36" / "26 to 36"
_GRAIN_RANGE_HEAD_RE = re.compile(
    r"\d\s*[–—−-]\s*$|\d\s+(?:to|through)\s+$", re.IGNORECASE)  # "36" in "26–36" / "26 to 36"

def _grain_is_range_endpoint(text: str, start: int, end: int) -> bool:
    return bool(_GRAIN_RANGE_TAIL_RE.match(text[end:end + 16])) or bool(
        _GRAIN_RANGE_HEAD_RE.search(text[max(0, start - 16):start]))
```

```python
def _grain_products(pair_counts, ks):
    cands = {p: str(p) for p in pair_counts}
    for a, b in itertools.combinations(sorted(pair_counts), 2):
        cands.setdefault(a + b, f"{a}+{b}")       # pooled pair SETS (the incident
                                                   # figure pools 180 fmt + 15 mq = 195)
    agg = {}                                       # product -> (repr, s) for the WARN text
    for p, rep in cands.items():
        for s in (1, 2, 3, 4):                     # scheme/arm multiplier; parsing the
            agg.setdefault(p * s, (rep, s))        # donor-scheme count from prose is
    draw = {m * k for m in agg for k in ks}        # fragile — a small-integer set is not
    return agg, draw

def _grain_near(n, m):                             # "≈" := within max(2, 2% of m)
    return abs(n - m) <= max(2, round(0.02 * m))   # 2 absorbs the incident's judge-drop
                                                   # attrition (388 vs 390); 2% scales it
```
