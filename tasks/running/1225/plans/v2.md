#### 4.2 Deliverable A — the lint check (`scripts/workflow_lint.py`)

New function, placed near the other single-file content pins (after `check_vm_thread_cap_guidance`, ~line 8400):

```python
_ASW_REL = "scripts/autonomous_session_watch.py"
_ASW_HEADER_RE = re.compile(r"^(\d+) passes\b", re.M)
_ASW_ITEM_RE = re.compile(r"^(\d+)\. \*\*", re.M)
# The crash-recovery respawn loop is an inline block in main() (not a
# *_pass function); it counts as one pass. If it is ever refactored into
# a crash_recovery_pass() function, set this to 0.
_ASW_INLINE_PASS_BLOCKS = 1


def check_asw_docstring_pass_count(*, watcher_path: Path | None = None) -> list[str]:
    """FAIL if the watcher docstring's '<N> passes' header digit diverges from
    the numbered inventory (``^\\d+. **`` items), the items are not exactly
    1..N sequential, or N diverges from the live pass set in ``main()``
    (distinct ``*_pass`` calls + the inline crash-recovery block).
    ``watcher_path`` is a unit-test override; production callers pass None.
    Bundled into the no-flags default run.  (#1225; manual catch-ups #1021,
    #1169.)"""
    path = watcher_path if watcher_path is not None else _REPO_ROOT / _ASW_REL
    errors: list[str] = []
    if not path.is_file():
        return [f"{path}: missing — cannot verify the watcher docstring pass count."]
    try:
        tree = ast.parse(path.read_text(encoding="utf-8"))
    except SyntaxError as exc:
        return [f"{path}: unparseable ({exc}) — cannot verify the docstring pass count."]
    doc = ast.get_docstring(tree, clean=False)
    if not doc:
        return [f"{path}: no module docstring — the pass inventory contract is gone."]

    # (1) header digit — exactly one '^<N> passes' line-start match required.
    headers = _ASW_HEADER_RE.findall(doc)
    if len(headers) != 1:
        errors.append(
            f"{path}: expected exactly one line-start '<N> passes' digit header in the "
            f"module docstring, found {len(headers)} — the count must be a DIGIT "
            f"('24 passes'), never a number word ('Fourteen passes'); see #1225."
        )
        return errors  # items/live checks are meaningless without a parseable header
    header_n = int(headers[0])

    # (2) numbered items — exactly 1..N, sequential, no holes/duplicates.
    items = [int(m.group(1)) for m in _ASW_ITEM_RE.finditer(doc)]
    if items != list(range(1, len(items) + 1)):
        errors.append(
            f"{path}: docstring numbered items are not exactly 1..{len(items)} "
            f"in order (got {items}) — renumber the inventory."
        )
    if len(items) != header_n:
        errors.append(
            f"{path}: docstring header says {header_n} passes but the inventory has "
            f"{len(items)} numbered items — add/remove the item AND update the digit."
        )

    # (3) live-pass cross-check: distinct *_pass calls inside main() + the
    # inline crash-recovery block must equal the header digit. This is the
    # assertion that would have caught #1021/#1169 (code->doc drift).
    main_fn = next(
        (n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == "main"), None
    )
    if main_fn is None:
        errors.append(f"{path}: no main() found — cannot cross-check the live pass set.")
        return errors
    live = {
        node.func.id
        for node in ast.walk(main_fn)
        if isinstance(node, ast.Call)
        and isinstance(node.func, ast.Name)
        and node.func.id.endswith("_pass")
    }
    live_n = len(live) + _ASW_INLINE_PASS_BLOCKS
    if live_n != header_n:
        errors.append(
            f"{path}: docstring header says {header_n} passes but main() calls "
            f"{len(live)} distinct *_pass functions (+{_ASW_INLINE_PASS_BLOCKS} inline "
            f"crash-recovery block) = {live_n} — a pass was added/removed without "
            f"reconciling the docstring inventory (add a numbered item + bump the "
            f"digit), or the inline-block constant _ASW_INLINE_PASS_BLOCKS is stale."
        )
    return errors
```

Notes the implementer must keep: (a) `ast.get_docstring` — not a raw-text regex over the whole file — so numbered lists in comments/other string constants can never false-match (the brief's robustness requirement); (b) the distinct-name SET dedupes the `--*-only` ladder's duplicate calls (e.g. `auth_outage_pass` appears at line ~19507-in-ladder AND 19573); (c) `subfloor_sentinel_pass` is excluded naturally because the walk is scoped to `main_fn`, and it is only called inside `vm_disk_pass`; (d) the early `return` after a failed header parse is deliberate — the word-form regression fails LOUD with one clear message.

### 11. Decision Rationale

- **Digit-form header rewrite ("24 passes") vs parsing number words.** *What:* rewrite the header to a line-start digit; the check refuses word forms. *Why:* a number-word parser needs a lookup table with an arbitrary cap and fails silently past it; a digit regex is trivial, and a word-form regression fails LOUD (0 header matches → error). *Source:* the task Goal's own phrasing ("regex the header int"); sibling precedent `check_vm_thread_cap_guidance` (workflow_lint.py:8307) pins a literal machine-readable form rather than parsing prose. *Alternatives:* word-number table (rejected: brittle, capped); accepting both forms (rejected: two spellings of one contract invites drift).
