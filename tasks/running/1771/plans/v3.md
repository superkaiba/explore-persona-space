def _resolve_autonomous_plan_gate(gpu_hours: float | None) -> tuple[str, bool]:
    """... GPU-HOUR-BLIND as of #1771 (user directive 2026-07-28): an
    autonomous session auto-approves ANY plan carrying a parseable GPU-hour
    estimate, regardless of magnitude. EPM_PLAN_AUTOAPPROVE_GPU_HOURS is no
    longer read. FAIL SAFE (retained): a missing/None gpu_hours parks —
    a correctness guard against an unestimated plan, not a cost control. ..."""
    _auto_raw = os.environ.get("EPM_AUTONOMOUS_SESSION", "").strip().lower()

    if gpu_hours is None:
        return ("parked_no_estimate", True)
    return ("auto_approved", True)
