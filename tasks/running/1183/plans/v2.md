```python
def _resolve_live_sidecar(*, seed: Path, override: Path, filename: str,
                          bootstrap: bytes, label: str) -> Path:
    """Shared #821 lazy resolver: honor a monkeypatched override; else
    <git-common-dir>/eps/<filename>, migrating seed→live atomically under
    locked_pods_conf on first use; read-only-FS loud-WARN seed fallback.
    Never called at module import time."""
    if override != seed:                      # test monkeypatch honored
        return override
    try:
        common = _git_common_dir()
    except RuntimeError:
        return seed                            # no-git fallback (tarball)
    live = common / _LIVE_PODS_CONF_DIRNAME / filename
    if live.exists():
        return live                            # steady-state fast path
    with locked_pods_conf():                   # reentrant — safe when held
        if live.exists():
            return live
        try:
            live.parent.mkdir(parents=True, exist_ok=True)
        except OSError as exc:
            print(f"[pod_config] WARN: cannot create live {label} dir ... ", file=sys.stderr)
            return seed
        payload = seed.read_bytes() if seed.exists() else bootstrap
        tmp = live.with_suffix(live.suffix + ".tmp")
        try:
            tmp.write_bytes(payload); os.replace(tmp, live)
        except OSError as exc:
            with contextlib.suppress(FileNotFoundError): tmp.unlink()
            print(f"[pod_config] WARN: could not migrate {label} → {live} ...", file=sys.stderr)
            return seed
        print(f"[pod_config] migrated {seed} → {live} (live {label} now lives OUT of git's blast radius ...)", file=sys.stderr)
        return live

def _resolve_live_pods_conf() -> Path:
    return _resolve_live_sidecar(seed=PODS_CONF_SEED, override=PODS_CONF,
        filename=_LIVE_PODS_CONF_FILENAME, bootstrap=_PODS_CONF_BOOTSTRAP_HEADER,
        label="pods.conf")

def resolve_live_pods_ephemeral() -> Path:
    """LIVE pods_ephemeral.json path (task #1183; mirrors #821 pods.conf)."""
    return _resolve_live_sidecar(seed=PODS_EPHEMERAL_SEED, override=PODS_EPHEMERAL_JSON,
        filename=_LIVE_PODS_EPHEMERAL_FILENAME, bootstrap=_PODS_EPHEMERAL_BOOTSTRAP,
        label="pods_ephemeral.json")
```

def _set_manual_override(pod_name: str, *, value: bool) -> str | None:
    # Monkeypatch fast path checked BEFORE the lock (consistency-checker r1):
    # a patched PODS_EPHEMERAL_JSON must never touch the real shared lockfile.

            data = json.loads(path.read_text())
        except json.JSONDecodeError as exc:
            print(f"WARNING: {path} JSON parse error: {exc}; ...", file=sys.stderr)
            return None

            return None

            return f"pods_ephemeral.json: manual_override for {pod_name} already {value}"
        pods[pod_name]["manual_override"] = value
        _atomic_write_text(path, json.dumps(data, indent=2) + "\n", default_mode=0o644)

                         *, allow_remove: frozenset[str] | set[str] = frozenset()) -> None:
    """... Atomic (tmp+os.replace) and guarded: an incoming dict that DROPS
    on-disk entries not named in allow_remove gets them re-added with a loud
    WARN (structural never-drop guard, task #1183 — no live-API call; the
    legitimate removers pass allow_remove explicitly)."""

## 4. Controls → Compatibility analysis (callers, worktrees, concurrent sessions)

- **Worktrees:** the live path keys off `_git_common_dir()` — identical across every worktree (shared `.git`), so all checkouts converge on one file, subsuming the #500 divergence class for the JSON just as #821 did for pods.conf. - **Concurrent sessions:** all JSON read-modify-writes serialize under the same flock as pods.conf ops (`~`-free, main-repo `scripts/.pods.conf.lock`, `pod_config.py:340`); flock auto-releases on process death.
