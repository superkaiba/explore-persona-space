ISSUE_RE = re.compile(r"issue[_-]?(\d{1,5})", re.IGNORECASE)

PROBE_TITLES = ("quota probe (auto-deleted)", "remove quota probe")

def parse_issue_id(path: str) -> int | None:
    for seg in path.split("/")[:2]:          # first two segments only
        if (m := ISSUE_RE.search(seg)):
            return int(m.group(1))
    return None

    # "probe" | "probe-cleanup" | "upload" | "other"  (exact-title probe match;
    # "upload" = any non-probe commit whose title indicates content landing,
    # e.g. startswith "Upload"; everything else "other" — all counted, none dropped)

def aggregate(entries: list[RepoFile], registry: dict) -> dict:
    # 1. top-level rollup: first path segment -> {files, bytes, lfs_bytes}
    # 2. issue rollup: parse_issue_id -> {prefixes, files, bytes, lfs_bytes,
    #       task_status, task_kind, has_clean_result, classification|None,
    #       ladder: {rung_dirs, rung_files, rung_bytes, prunable_rungs(=all but
    #       highest-numbered rung per parent, heuristic)}}

    # 4. overflow pointers: canonical entries named OVERFLOW_POINTER.json ->
    #       count + prefixes (+ parse ≤ max_pointer_downloads via hf_hub_download)
    # 5. INTERNAL CONSISTENCY (fail loud): sum(rollup files) == len(entries);
    #       cross-check len(entries) vs len(list_repo_files_complete(...)) >= 0.99

def scan_commits(api, repo_id, since: date) -> dict:
    # list_repo_commits newest-first; stop past `since`; classify each;
    # -> {n_probe, n_upload, n_other, first_nonprobe_upload: {id, date, title},
    #     per_day_upload_counts}

def main():
    for repo in repos:
        entries = list_repo_tree_complete(api, repo)
        report = aggregate(entries, registry)
        report["commits"] = scan_commits(api, repo, since)
        write_json(out_dir / f"hf_repo_audit_{slug}.json", report)

## 11. Decision Rationale

- **`expand=False` for the tree walk.** What: enumerate at 1000 entries/page with `size` + `lfs` per `RepoFile`. Why: `expand=True` drops to 50/page (20× the requests) to fetch `last_commit`/security info the audit doesn't need. Source: installed `huggingface_hub` 0.36.2 `HfApi.list_repo_tree` docstring (read this session; the docstring's `RepoFile(path=..., size=..., lfs=...)` example is at default expand). Alternatives: `expand=True` (kill-criterion 3 fallback only); `get_paths_info` (per-path POST — wrong tool for full enumeration).
- **Transient-retry via `hub._retry_upload`.** Why: `huggingface_hub.paginate` retries only 429 on cursor pages; 504s propagate (gotchas, #794/#658). Source: `hub.py::list_repo_files_complete` docstring + #794/#658. Alternatives: hand-rolled retry (rejected — module standard exists).
- **Pointer-download cap = 50.** Source: ungrounded — harmless (bounds wall time; pointers beyond the cap are counted, not parsed; expected total is tens at most).
- **Commit-scan cutoff 2026-07-06.** One day before the 2026-07-07 rejection, capturing the boundary. Source: #1090/#1108 incident timestamps. Harmless — widening the window only adds pages.
- **Ladder-rung regex** `(checkpoint|step|rung|epoch|ckpt)[-_]?\d+`. Source: `trainer.py` `ignore_patterns=["checkpoint-*"]` + #1090's per-rung ladder naming; explicitly a HEURISTIC — its rollups are reported as estimates, never gated on.
- **99% consistency threshold.** Source: ungrounded — harmless (expect 100%; slack for in-flight commits between the two reads).
- N/A — no model training; no other hyperparameters exist in this task type.
