```python
# ── Quota-exhausted sentinel (#1126) ─────────────────────────────────
# On a hard org usage-limit terminal (job spawns, dies phase=failed in
# ~30s, fetched result text = "You've hit your usage limit ... try
# again at Aug 6th, 2026 6:26 AM"), write a durable shared sentinel and
# short-circuit every later dispatch until the parsed reset timestamp.
# Sentinel lives under DISPATCH_ROOT (main checkout, worktree-safe);
# .claude/cache/ is gitignored (verified: .gitignore line 78).
QUOTA_SENTINEL_RELPATH = Path(".claude/cache/codex-quota-exhausted-until")
EXIT_QUOTA_EXHAUSTED = 9  # free: 0-8, 130, 143 taken; NOT in TRANSIENT_FAIL_EXIT_CODES
QUOTA_LIMIT_RE = re.compile(r"hit your usage limit", re.IGNORECASE)
# "try again at Aug 6th, 2026 6:26 AM" — month day(ordinal), year[, h:mm AM/PM]
QUOTA_RESET_RE = re.compile(
    r"try again at\s+([A-Za-z]{3,9})\.?\s+(\d{1,2})(?:st|nd|rd|th)?,?\s+(\d{4})"
    r"(?:,?\s+(\d{1,2}):(\d{2})\s*([AP])\.?M\.?)?",
    re.IGNORECASE,
)
QUOTA_FALLBACK_TTL_SECS = 24 * 3600  # parse failure never wedges dispatch forever
QUOTA_MAX_PLAUSIBLE_SECS = 45 * 24 * 3600  # a parse beyond ~45d is treated as a parse failure
```

Notes: the detection regex is apostrophe-free (`You've` could be ASCII `'` or U+2019 in future CLI versions — the substring `hit your usage limit` sidesteps it) and deliberately does NOT match the pre-existing `codex app-server exited unexpectedly` transient class.

### 4b. New helpers — insert before the `# Main lifecycle` banner (line ~725)

```python
def _quota_sentinel_path() -> Path:
    """Sentinel path: env override (tests/ops) > DISPATCH_ROOT-anchored default."""
    override = os.environ.get("EPM_CODEX_QUOTA_SENTINEL_PATH")
    if override:
        return Path(override)
    return DISPATCH_ROOT / QUOTA_SENTINEL_RELPATH


def _parse_quota_reset(text: str) -> float | None:
    """Parse 'try again at Aug 6th, 2026 6:26 AM' -> unix epoch (VM-local
    time; the Codex CLI renders local time and states no zone). Returns
    None when absent, unparseable, in the past, or implausibly far out
    (> QUOTA_MAX_PLAUSIBLE_SECS) — caller falls back to the 24h TTL."""
    m = QUOTA_RESET_RE.search(text)
    if not m:
        return None
    month_s, day_s, year_s, hh, mm, ap = m.groups()
    try:
        month = datetime.datetime.strptime(month_s[:3].title(), "%b").month
        hour, minute = 0, 0
        if hh is not None:
            hour = int(hh) % 12
            if ap and ap.upper() == "P":
                hour += 12
            minute = int(mm)
        ts = time.mktime(
            datetime.datetime(int(year_s), month, int(day_s), hour, minute).timetuple()
        )
    except (ValueError, OverflowError):
        return None
    now = time.time()
    if ts <= now or ts > now + QUOTA_MAX_PLAUSIBLE_SECS:
        return None
    return ts


def _write_quota_sentinel(result_text: str, job_id: str | None) -> str | None:
    """Atomically (tmp + os.replace, same dir) write the sentinel. Returns
    the until-ISO string on success, None on any failure. FAIL-SOFT: never
    raises — a sentinel-write failure must not mask the original failure path."""
    try:
        until_unix = _parse_quota_reset(result_text)
        parse_ok = until_unix is not None
        if until_unix is None:
            until_unix = time.time() + QUOTA_FALLBACK_TTL_SECS
        until_iso = datetime.datetime.fromtimestamp(
            until_unix, datetime.timezone.utc
        ).isoformat(timespec="seconds")
        payload = {
            "until_unix": until_unix,
            "until_iso": until_iso,
            "parse_ok": parse_ok,
            "detected_at_iso": datetime.datetime.now(datetime.timezone.utc).isoformat(
                timespec="seconds"
            ),
            "job_id": job_id,
            "raw_excerpt": result_text[:300],
        }
        path = _quota_sentinel_path()
        path.parent.mkdir(parents=True, exist_ok=True)
        fd, tmp = tempfile.mkstemp(dir=str(path.parent), prefix=".codex-quota-tmp-")
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as fh:
                fh.write(json.dumps(payload, indent=2))
            os.replace(tmp, path)
        finally:
            Path(tmp).unlink(missing_ok=True)  # no-op after a successful replace
        print(
            f"codex-quota-exhausted: wrote sentinel {path} "
            f"(until {until_iso}, parse_ok={parse_ok})",
            file=sys.stderr,
        )
        return until_iso
    except Exception as exc:
        print(f"WARN: could not write quota sentinel: {exc}", file=sys.stderr)
        return None


def _quota_sentinel_active() -> str | None:
    """Return the until-ISO string when a valid sentinel is in the FUTURE,
    else None. Expired or corrupt content -> best-effort delete + None
    (FAIL-OPEN toward dispatching: a bad sentinel must never wedge Codex
    off permanently). A read error (other than missing) -> WARN + None."""
    if os.environ.get("EPM_SKIP_CODEX_QUOTA_SENTINEL") == "1":
        return None
    path = _quota_sentinel_path()
    try:
        raw = path.read_text(encoding="utf-8")
    except FileNotFoundError:
        return None
    except OSError as exc:
        print(f"WARN: quota sentinel unreadable ({exc}); ignoring.", file=sys.stderr)
        return None
    try:
        data = json.loads(raw)
        until_unix = float(data["until_unix"])
        until_iso = str(data.get("until_iso") or until_unix)
    except Exception:
        print(
            f"WARN: corrupt quota sentinel {path}; deleting and proceeding.",
            file=sys.stderr,
        )
        try:
            path.unlink(missing_ok=True)
        except OSError:
            pass
        return None
    if until_unix <= time.time():
        try:
            path.unlink(missing_ok=True)  # self-expiry
        except OSError:
            pass
        return None
    return until_iso
```

### 4d. Short-circuit — in `main()`, insert AFTER the reattach block (after line 1338 `return _run_reattach(companion, args)`), BEFORE `# Resolve prompt.` (line 1340)

```python
    # Quota-exhausted short-circuit (#1126): while the shared sentinel is in
    # the future, exit fast with the SAME terminal-marker contract the
    # orchestrator's no-show fallback already reads — no spawn, no poll, no
    # 30s wait. Reattach (above) deliberately BYPASSES this: attaching to an
    # EXISTING job spends no new quota and may still harvest a completed
    # result. Manual override: delete the sentinel file (or set
    # EPM_SKIP_CODEX_QUOTA_SENTINEL=1).
    quota_until = _quota_sentinel_active()
    if quota_until is not None:
        return _fail(
            args.issue,
            None,
            (
                f"codex-quota-exhausted: org usage limit exhausted until "
                f"{quota_until}; dispatch short-circuited (no Codex job "
                f"spawned). Sentinel: {_quota_sentinel_path()} — delete it "
                f"to force a probe dispatch."
            ),
            EXIT_QUOTA_EXHAUSTED,
        )
```

Properties, each mapped to a design requirement:
- **Same external contract:** `_fail` posts `epm:codex-task-failed` (only with `--issue`) and prints the loud `ERROR:` line — the orchestrator's existing no-show fallback fires unchanged; `epm:codex-task-spawned` is NOT posted (nothing spawned; the check sits before prompt resolution and `_resolve_companion`).
- **Exit 9 is distinct:** ∉ {0,1,2,3,4,5,6,7,8,130,143}, ∉ `TRANSIENT_FAIL_EXIT_CODES` → the transient re-dispatch loop (lines 1382–1399) can never fire on it (the check is also before the attempt loop entirely, so the loop is never entered).
- **Marker noise is bounded and intentional:** one 1-line `epm:codex-task-failed` per round per session — the per-round no-show record the orchestrator reads.
- **Reattach bypass:** placement after the `if args.reattach:` early-return makes the bypass structural. Justification: a reattach targets an already-spawned job (already-spent quota); its result may already be terminal and fetchable — blocking that harvest would strand a paid-for output.
- **Why code, not a model call:** this is deterministic control flow in a dispatch helper (regex + timestamp + file IO); a model call adds latency, cost, and nondeterminism for zero benefit.

1. **Sentinel path `.claude/cache/codex-quota-exhausted-until` under `DISPATCH_ROOT`.** Why: shared across all sessions AND all worktrees (codex_task.py is dispatched from worktree copies too; `DISPATCH_ROOT` already resolves the main checkout via `git rev-parse --git-common-dir`, lines 134–178); `.claude/cache/` is the project's established durable-VM-local-state dir (disk-guard/gcp-janitor/auth-outage sidecars) and is gitignored (`.gitignore:78`, verified via `git check-ignore`). Alternatives: `~/.eps-autonomous/` (watcher state dir — wrong owner; this is a dispatch-helper concern), `/tmp` (not durable across reboots; swept by janitors), `PROJECT_ROOT` (worktree-relative — the exact bug the brief warns about).
2. **Exit code 9.** Why: the full taxonomy 0–8 + 130/143 is occupied (module docstring lines 56–88 + code); 9 is the smallest free code; it must be ∉ `TRANSIENT_FAIL_EXIT_CODES = {3,4,5,8}` (line 200) so the re-dispatch loop never retries a quota short-circuit, and ∉ {0,1,2} (success/terminal-fail/argparse). Source: codex_task.py lines 180–240, 1382–1399. Alternatives: reuse 1 (loses the distinct machine-readable signal the Goal asks for), ≥64 (no benefit, breaks the compact taxonomy).
3. **Detection point = `_finalize_result`, `phase == "failed"` + fetched-stdout regex.** Why: the ONLY observed manifestation (evidence §2: spawn succeeds, `terminal phase=failed after 30s`, fetch rc=0, limit text IS the fetched stdout — 15+ 317 B `/tmp` artifacts + issue 1122/1124 markers, 2026-07-08); `stdout` is in scope only there; the path is shared with `--reattach`, covering the harvest case for free. Alternatives: spawn-stderr matching (no evidence the error surfaces there — spawn succeeds), phase-agnostic text match (risks false-positive on a hypothetical done-phase critique quoting the string; scoped to failed-phase where a match is unambiguous).
4. **Detection regex `hit your usage limit` (case-insensitive).** Why: verbatim substring of the observed error, apostrophe-free (robust to ASCII `'` vs U+2019 drift), does not match the sibling `app-server exited unexpectedly` transient class. Source: the verbatim artifact text (§2). Alternatives: full-sentence match (brittle to CLI copy edits), matching `try again at` alone (too generic).
5. **Reset-timestamp parse as VM-local naive time + plausibility window; fallback TTL 24h.** Why: the CLI states no timezone ("Aug 6th, 2026 6:26 AM"); VM is `America/Los_Angeles` (verified `date` → PDT) and the incident timeline (exhausted ~17:00Z = 10:00 PDT Jul 7 → reset Aug 6 6:26 AM ≈ a 30-day window) is consistent with local rendering; mis-zone costs ≤ hours and early expiry self-heals (the next failure re-writes with the still-future date). Parse failure → now+24h so a malformed date never wedges dispatch forever (the brief's requirement); past / >45d parses are treated as failures. Source: `date` output; incident timeline in task body. Alternatives: assume UTC (no better grounded, same self-heal bound), no fallback TTL (a permanent unparseable sentinel = fail-closed, banned).
6. **Short-circuit posts `epm:codex-task-failed` (existing kind) with a `codex-quota-exhausted` note token; no new marker kind; no output-file write.** Why: the orchestrator no-show fallback already keys on this marker + the absence of a durable verdict in the output file (SKILL.md ~2611, ~5672) — zero orchestrator prose changes needed for correct behavior; NOT writing the output file avoids clobbering a possible prior-round verdict the Step 5b durable-verdict-first rule may still read. Alternatives: new `epm:codex-quota-exhausted` marker kind (a marker-schema addition → architectural review for no consumer benefit), writing an explanation to the output file (risks the stale-verdict clobber).
7. **Reattach bypasses the short-circuit.** Why: reattach targets an ALREADY-spawned job — quota already spent; the job may be terminal with a fetchable (pre-outage) result; blocking the harvest strands a paid-for output. Alternative: block reattach too (strands recoverable outputs, no quota saved).
8. **Fail-open posture everywhere (corrupt → delete + proceed; write/read errors → WARN + original path).** Why: the failure the sentinel prevents costs ~30s/round; a fail-closed bug would silently disable the entire Codex ensemble indefinitely — strictly worse (same asymmetry as the watcher's auth-outage guard). Alternative: monkeypatching `DISPATCH_ROOT` in tests (touches unrelated marker-post cwd behavior).
