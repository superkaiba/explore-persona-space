```python
_GIT_SIDE_ROOTS = frozenset({

# "HF" is case-sensitive (lowercase "hf" is a CLI name); the rest scoped-(?i).
_HF_CUE_RE = re.compile(r"\bHF\b|(?i:hugging\s*face|huggingface\.co|\bhub\b|\bdata[- ]repo\b)")
_HF_ISSUE_PREFIX_RE = re.compile(r"issue\d+_")  # HF layout `issue<N>_…` (git uses `issue_<N>`)
_UNPINNED_CUE_WINDOW = 250  # chars of same-line look-back for the cue
# A preceding backtick PARENT anchor (`issue<N>_…/`) for resolving a RELATIVE
# sub-path claim; bound like the Pattern-D binder (same line, bracket-free
# gap ≤ _SUBPATH_CLAIM_MAX_GAP).
_BACKTICK_ISSUE_PARENT_RE = re.compile(r"`(?P<parent>issue\d+_[A-Za-z0-9_\-./]{0,118}/)`")
# Per-body cap on unique main-revision count probes (same worst-case per-probe
# arithmetic as _HF_COUNT_MAX_PROBES, ≈22.5 s worst case each; unpinned claims
# are rare so 4 suffices — past-cap claims surface as unverified notes).
_HF_UNPINNED_MAX_PROBES = 4
```

```python
def check_hf_unpinned_count_claims(body: str) -> CheckResult:
    name = "backtick HF-path count claims carry an adjacent pinned /tree link"
    claims = _gather_hf_unpinned_count_claims(body)
    if not claims:
        return CheckResult(name, True, "no unpinned backtick HF-path count claims")
    # Check-level fence: unlike checks 30/32 (whose WARN needs a probe), the
    # missing-pin finding is text-derived — going fully quiet under the fence
    # keeps the offline suite (conftest sets EPM_VERIFY_BODY_NO_HF=1) and
    # grandfathered CI contexts byte-stable. Probe-level fail-soft is still
    # inherited from _hf_file_count_for_prefix on the online path.
    if os.environ.get("EPM_VERIFY_BODY_NO_HF") == "1" or not _hf_hub_importable():
        notes = "; ".join(f"`{tok}` ({c} {n})" for c, n, tok, _ in claims)
        return CheckResult(name, True,
            f"{len(claims)} unpinned claim(s) unverified (pin check needs HF access): {notes}")
    warn_lines: list[str] = []
    probe_memo: dict[tuple[str, str, str, str], tuple[str, int, int, str]] = {}
    for count, noun, tok, resolved in claims:
        base = (f"body claims {count} {noun} at backtick `{tok}` with no adjacent "
                f"pinned /tree/<sha> link")
        if resolved is None:
            warn_lines.append(base + " (not resolvable against the data repo at main — "
                              "no `issue<N>_…/` anchor)")
            continue
        key = (_HF_DATA_REPO_ID, _HF_DATA_REPO_TYPE, "main", resolved.strip("/"))
        # memo + cap: same closure shape as check 30's _probed (:9067-9080),
        # cap _HF_UNPINNED_MAX_PROBES; past-cap → "(per-body probe cap)" note.
        status, n_files, n_dirs, note = ...  # via memoized _hf_file_count_for_prefix
        if status != "ok":
            warn_lines.append(base + f" (count not confirmed at main: {note})")
        elif count == n_files or (noun.lower().startswith("shard") and count <= n_files):
            warn_lines.append(base + f" — `{resolved}` at `{_HF_DATA_REPO_ID}@main` holds "
                              f"{n_files} file(s) (count consistent; add a pinned /tree/<sha> link)")
        else:
            msg = base + (f" — and `{resolved}` at `{_HF_DATA_REPO_ID}@main` holds "
                          f"{n_files} file(s), not {count}")
            if n_dirs and count == n_files + n_dirs:
                msg += (f" + {n_dirs} folder(s) — the claimed count is consistent with "
                        "files+folders (folder entries are not files)")
            elif count < n_files:
                msg += " (or the claim describes a subset of the prefix)"
            warn_lines.append(msg)
    return CheckResult(name, True, "; ".join(warn_lines), is_warn=True)
```

Key semantics decisions (each with rationale):
- **WARN-only, no `passed=False` path anywhere** — hard constraint from the task body; identical to check 30 (`:9025-9027`).
- **Online, the missing-pin WARN fires for every extracted claim regardless of probe outcome.** The pin's absence is a fact about the body text, not the network; a 429/not_found/cap only downgrades the count-resolution HALF of the line to an unverified note, mirroring check 30's "unverified appended to the WARN detail, never dropped" rule (`:9044-9046`). A wrong parent-anchor join that lands on a nonexistent path is `not_found` → skip → note, never a spurious mismatch text.
- **Fence / missing `huggingface_hub` → check-level PASS + note** (per the task body's fail-soft list and the brief's test (c)); `_hf_hub_importable()` is a 3-line try/except-import helper (or inline try/except, implementer's choice).
- **`main` is a moving ref**: `_hf_file_count_for_prefix`'s success-only cache keys include the revision string, so a `main` listing is cached per-process only — acceptable for the short-lived verifier CLI; document the staleness caveat in the docstring.
- **Probe budget** own memo, cap 4, shared `_HF_TREE_FILE_COUNT_CACHE` (already cleared by the test fixture at `tests/test_verify_task_body.py:1687-1703` — no fixture change needed).
