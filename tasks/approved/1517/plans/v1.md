```python
def _stage_tail_into(bundle_dir, name, src):
    # #1517: tail-capped staging for the canonical workload log. upload_folder
    # force-routes any >10 MB blob to LFS regardless of extension, and over the
    # LFS quota that upload silently fails (the #1481 shape: manifests landed,
    # the log alone did not) — the traceback is at the END of a log (#885/#935),
    # so stage the LAST LOG_FILE_CAP bytes under the SAME repo name (paths stay
    # byte-identical, the #1151 contract). Failures logged, never raised.
    try:
        bundle_dir.mkdir(parents=True, exist_ok=True)
        size = Path(src).stat().st_size
        kept = min(size, LOG_FILE_CAP)
        with open(src, "rb") as fin, open(bundle_dir / name, "wb") as fout:
            if size > LOG_FILE_CAP:
                fin.seek(size - LOG_FILE_CAP)
            fout.write(fin.read(LOG_FILE_CAP))
        if size > LOG_FILE_CAP:
            _say(f"[crash-persist] staged {name} (last {kept} of {size} bytes)")
        else:
            _say(f"[crash-persist] staged {name} ({(bundle_dir / name).stat().st_size} bytes)")
    except Exception as exc:
        _say(f"[crash-persist] FAILED staging {name}: {exc}")
```
