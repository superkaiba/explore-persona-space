```python
def _c71_run_bounded(argv, input_bytes, timeout_s):
    """Run jq with REAL pre-materialization bounds (#2590 MF-A). Returns
    (status, rc, stdout_bytes, stderr_bytes); status in {"ok", "timeout",
    "output-cap", "launch-error"}. NEVER raises (A14: verify_plan_text has
    no per-check exception containment); the child is reaped on every path.
    capture_output/communicate() would materialize the child's ENTIRE stdout
    before any truncation — jq's output-expanding constructs (while/repeat/
    recurse) need none of the refused flags and emit hundreds of MB inside
    the timeout — so the pumps read incrementally and the parent KILLS the
    child the moment either stream crosses its cap."""
    try:
        proc = subprocess.Popen(
            argv, stdin=subprocess.PIPE, stdout=subprocess.PIPE,
            stderr=subprocess.PIPE, cwd=str(_C71_REPO_ROOT), env=_C71_JQ_ENV)
    except (OSError, subprocess.SubprocessError):
        return ("launch-error", None, b"", b"")   # MF-A item 2 → whole-check SKIP at the call site
    except ValueError:
        return ("launch-error", None, b"", b"")   # argv content (e.g. embedded NUL) → skip-instance
    bufs = {"out": bytearray(), "err": bytearray()}
    capped = threading.Event()

    def _pump(stream, key, cap):
        try:
            while True:
                chunk = stream.read(65536)
                if not chunk:
                    return
                bufs[key].extend(chunk[: cap + 1 - len(bufs[key])])
                if len(bufs[key]) > cap:
                    capped.set()      # stop reading; the kill below unblocks the child
                    return
        except (OSError, ValueError):
            return

    def _feed():
        try:
            proc.stdin.write(input_bytes)
            proc.stdin.close()
        except (OSError, ValueError):
            pass                      # BrokenPipe when the child exits early: expected

    threads = [
        threading.Thread(target=_feed, daemon=True),
        threading.Thread(target=_pump, args=(proc.stdout, "out", _C71_MAX_OUTPUT_BYTES), daemon=True),
        threading.Thread(target=_pump, args=(proc.stderr, "err", _C71_MAX_STDERR_BYTES), daemon=True),
    ]
    for t in threads:
        t.start()
    deadline = time.monotonic() + timeout_s
    status = "ok"
    while proc.poll() is None:
        if capped.is_set():
            status = "output-cap"
            break
        if time.monotonic() >= deadline:
            status = "timeout"
            break
        time.sleep(0.05)
    if proc.poll() is None:
        proc.kill()
    try:
        proc.wait(timeout=5)
    except subprocess.TimeoutExpired:
        status = "timeout"            # unkillable child: report, never raise
    for t in threads:
        t.join(timeout=1.0)
    return (status, proc.returncode if status == "ok" else None,
            bytes(bufs["out"]), bytes(bufs["err"]))
```

```python
    r"(?i)(?:vllm_)?max_model_len\s*[=:]\s*"
    r"([A-Za-z_][\w.]*(?:\s*[+\-*/]\s*[\w.]+)+"          # form 1: expr with >=1 operator (budget+GEN_MAX)
    r"|[A-Z][A-Z0-9_]{2,}|[a-z][\w]*_[\w]+)"             # form 2: single CONSTANT-style identifier
)
# MF-C: the re-pin satisfier is LINE-scoped with regen co-reference — a whole-plan
# re.search let a 'dedicated … engine' ANYWHERE (an unrelated embedding or
# primary-inference stage) suppress the WARN, and the [^.;] gap classes MATCH
# NEWLINES, so even stage-anchored alternatives cross-matched document-wide.
_C69_REPIN_REGEN_TOK  = re.compile(r"(?i)\bre-?gen\w*\b|\bre-?pin\w*\b")
_C69_REPIN_OBJ_TOK    = re.compile(r"(?i)\bmax_model_len\b|\bengine\b")
_C69_REPIN_ACT_STRONG = re.compile(
    r"(?i)[≥>]=?|\bsized?\b|\bre-?siz\w+|\bre-?instantiat\w+|\bdedicated\b|2\s*[×x]\b|\bre-?pin\w*\b"  # noqa: RUF001
)
_C69_REPIN_ACT_WEAK   = re.compile(r"(?i)\bfits?\b")   # weak: requires max_model_len on the line
_C69_MML_RE           = re.compile(r"(?i)\bmax_model_len\b")

def _c69_repin_lines(plan: str) -> list[tuple[int, str]]:
    """Line-scoped regen-time re-pin satisfier (#2590 MF-C). A line satisfies
    iff it carries (1) regen/re-pin co-reference, (2) an engine-length object
    token, and (3) a sizing/action token — strong action tokens pair with
    either object; the weak token ('fit(s)', demonstrated compliant prose,
    #1005 v4:118) requires the explicit max_model_len object, because 'fit'
    is ubiquitous ridge-fit vocabulary in this corpus. RAW lines, fences
    included (the c69 pin-scan convention)."""
    hits: list[tuple[int, str]] = []
    for i, line in enumerate(plan.splitlines(), start=1):
        if not _C69_REPIN_REGEN_TOK.search(line):
            continue
        if _C69_REPIN_OBJ_TOK.search(line) and _C69_REPIN_ACT_STRONG.search(line):
            hits.append((i, line.strip()))
        elif _C69_MML_RE.search(line) and _C69_REPIN_ACT_WEAK.search(line):
            hits.append((i, line.strip()))
    return hits
```
