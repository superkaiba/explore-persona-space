### 3a. `scripts/codex_daemon_reaper.py` (new, ~150 lines)

```
#!/usr/bin/env python3
"""Reap leaked Codex app-server daemons + truncate the codex log WAL.

The Codex ensemble-review path (scripts/codex_task.py -> openai-codex plugin)
spawns a persistent daemon trio per session (node codex app-server, its
codex-linux-x64 vendor binary, app-server-broker.mjs serve) that does NOT exit
after the companion task completes. They accumulate over weeks, each holding
~/.codex/logs_2.sqlite (WAL mode) open so SQLite can never checkpoint the WAL.

Mirrors scripts/cron_pod_audit.sh / scripts/worktree_audit.py: report-only by
default, --apply to act, --json summary, exit 2 when something was (would be)
reaped so the cron wrapper can swallow it.
"""
import argparse, json, os, re, signal, sqlite3, subprocess, sys, time
from pathlib import Path

DEFAULT_MAX_AGE_H = 24.0
CODEX_DB = Path(os.path.expanduser("~/.codex/logs_2.sqlite"))

# A process is a reap CANDIDATE iff its argv matches ANY of these AND it is
# older than the threshold AND it is not hard-excluded. Anchored on the real
# daemon argv shapes (confirmed live 2026-06-30), NOT on bare substrings that
# also appear inside shell-wrapper env/eval text (see HARD_EXCLUDE + R3).
DAEMON_MATCH = [
    re.compile(r"\bcodex app-server\b"),            # node ... codex app-server  AND  .../codex-linux-x64/.../bin/codex app-server
    re.compile(r"app-server-broker\.mjs\b"),        # node ... app-server-broker.mjs serve ...
    re.compile(r"\bcodex-linux-x64\b"),             # the vendor binary path
    re.compile(r"codex-companion\.mjs\b"),          # the companion runtime, if present
]

# Checked FIRST. A process whose argv contains ANY of these is NEVER reaped,
# regardless of age — even if it also matches a DAEMON_MATCH pattern. This is
# what protects the live codex_task.py review drivers (the bash -c wrapper +
# the uv/python children) AND unrelated workflow scripts.
HARD_EXCLUDE = [
    re.compile(r"codex_task\.py\b"),     # the review-dispatch drivers
    re.compile(r"\bworkflow_lint\b"),
    re.compile(r"\bwandb_reclaim\b"),
    re.compile(r"\bbash -c\b"),          # shell wrappers (their env text echoes 'codex-companion'/'app-server' — R3)
]


def _max_age_seconds(cli_h: float | None) -> float:
    """Threshold in seconds: CLI flag wins, else EPS_CODEX_REAPER_MAX_AGE_H, else 24h."""
    if cli_h is not None:
        return cli_h * 3600.0
    env = os.environ.get("EPS_CODEX_REAPER_MAX_AGE_H")
    return (float(env) if env else DEFAULT_MAX_AGE_H) * 3600.0


def _ps_snapshot() -> list[tuple[int, int, str]]:
    """One-shot (pid, etimes_seconds, argv) for every process. argv is the full
    command line (ps -o args), the field the selector matches — NOT comm, which
    reads 'node'/'codex'/'bash' for the daemons. Returns [] on ps failure
    (fail-soft: a reaper that can't read the table reaps nothing)."""
    try:
        out = subprocess.run(
            ["ps", "-eo", "pid=,etimes=,args="],
            capture_output=True, text=True, timeout=30,
        )
    except (OSError, subprocess.SubprocessError):
        return []
    rows = []
    for line in out.stdout.splitlines():
        parts = line.strip().split(None, 2)        # pid, etimes, "rest is argv"
        if len(parts) < 3:
            continue
        try:
            rows.append((int(parts[0]), int(parts[1]), parts[2]))
        except ValueError:
            continue
    return rows


def _is_daemon(argv: str) -> bool:
    return any(p.search(argv) for p in DAEMON_MATCH)


def _is_excluded(argv: str) -> bool:
    return any(p.search(argv) for p in HARD_EXCLUDE)


def enumerate_candidates(max_age_s: float, snapshot=None):
    """Return ({candidates}, {sub_threshold}) where each is a list of
    {pid, age_s, argv}. A candidate matches a daemon pattern, is NOT excluded,
    and is >= threshold. sub_threshold = matched + not-excluded but younger
    (reported so the dry-run shows the live daemons it is deliberately sparing).
    `snapshot` is injectable for tests (a list of (pid, etimes, argv) tuples)."""
    rows = snapshot if snapshot is not None else _ps_snapshot()
    candidates, sub_threshold = [], []
    for pid, age_s, argv in rows:
        if _is_excluded(argv) or not _is_daemon(argv):
            continue
        rec = {"pid": pid, "age_s": age_s, "argv": argv[:200]}
        (candidates if age_s >= max_age_s else sub_threshold).append(rec)
    return candidates, sub_threshold


# --- kill machinery: lifted from worktree_audit.py, re-verifying against the
#     reaper's OWN match+exclude predicate (not the worktree ORPHAN patterns). ---
def _read_cmdline(pid: int) -> str | None:
    try:
        with open(f"/proc/{pid}/cmdline", "rb") as fh:
            return fh.read().replace(b"\x00", b" ").decode("utf-8", "replace").strip()
    except OSError:
        return None


def _still_reapable(pid: int) -> bool:
    """PID-reuse guard: re-read /proc cmdline at signal time; only signal if it
    STILL matches a daemon pattern and is STILL not excluded (a recycled pid
    fails both and is spared)."""
    cmd = _read_cmdline(pid)
    return bool(cmd) and _is_daemon(cmd) and not _is_excluded(cmd)


def kill_candidates(pids, term_wait_s=8.0):
    """SIGTERM -> wait -> SIGKILL survivors; cmdline re-verified immediately
    before EVERY signal (R1). term_wait_s default 8s (R2). Returns (gone, leftover)."""
    pending = []
    for pid in pids:
        if not _still_reapable(pid):
            continue
        with __import__("contextlib").suppress(OSError):
            os.kill(pid, signal.SIGTERM)
        pending.append(pid)
    deadline = time.time() + term_wait_s
    while pending and time.time() < deadline:
        time.sleep(0.2)
        pending = [p for p in pending if _read_cmdline(p)]
    for pid in pending:
        if _still_reapable(pid):     # re-verify before escalating
            with __import__("contextlib").suppress(OSError):
                os.kill(pid, signal.SIGKILL)
    time.sleep(0.5)
    survivors = [p for p in pending if _read_cmdline(p)]
    gone = [p for p in pids if p not in survivors]
    return gone, survivors


def truncate_wal() -> dict:
    """Best-effort PRAGMA wal_checkpoint(TRUNCATE) on the codex DB. NEVER VACUUM
    (rewrites the whole multi-GB DB). A missing/locked file is reported, never
    raised (R4): if a daemon survived SIGKILL its open connection legitimately
    blocks truncation. Returns {ok, wal_bytes_before, wal_bytes_after, error}."""
    wal = CODEX_DB.with_name(CODEX_DB.name + "-wal")
    before = wal.stat().st_size if wal.exists() else 0
    if not CODEX_DB.exists():
        return {"ok": False, "error": "db_absent", "wal_bytes_before": before, "wal_bytes_after": before}
    try:
        con = sqlite3.connect(str(CODEX_DB), timeout=10.0)   # busy_timeout via timeout=
        try:
            con.execute("PRAGMA busy_timeout=10000")
            con.execute("PRAGMA wal_checkpoint(TRUNCATE)")
        finally:
            con.close()
    except sqlite3.Error as e:
        after = wal.stat().st_size if wal.exists() else 0
        return {"ok": False, "error": str(e), "wal_bytes_before": before, "wal_bytes_after": after}
    after = wal.stat().st_size if wal.exists() else 0
    return {"ok": True, "error": None, "wal_bytes_before": before, "wal_bytes_after": after}


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description="Reap leaked Codex app-server daemons + truncate the log WAL.")
    ap.add_argument("--apply", action="store_true", help="Actually kill + truncate (default: dry-run report).")
    ap.add_argument("--dry-run", action="store_true", help="Explicit report-only (default; mutually-informative with --apply).")
    ap.add_argument("--json", action="store_true", help="Emit a JSON summary.")
    ap.add_argument("--max-age-h", type=float, default=None,
                    help="Reap daemons older than this many hours (default: $EPS_CODEX_REAPER_MAX_AGE_H or 24).")
    args = ap.parse_args(argv)
    apply = args.apply and not args.dry_run     # --dry-run wins if both passed

    max_age_s = _max_age_seconds(args.max_age_h)
    candidates, sub_threshold = enumerate_candidates(max_age_s)

    gone, leftover, wal = [], [p["pid"] for p in candidates], None
    if apply and candidates:
        gone, leftover = kill_candidates([p["pid"] for p in candidates])
        wal = truncate_wal()                 # only meaningful after releasing connections
    elif apply:
        wal = truncate_wal()                 # no daemons over threshold, still reclaim WAL

    summary = {
        "apply": apply,
        "max_age_h": max_age_s / 3600.0,
        "n_candidates": len(candidates),
        "n_sub_threshold": len(sub_threshold),
        "candidates": candidates,
        "sub_threshold": sub_threshold,
        "killed": sorted(gone),
        "leftover": sorted(leftover),
        "wal": wal,
    }
    if args.json:
        print(json.dumps(summary))
    else:
        verb = "killed" if apply else "would reap"
        print(f"codex_daemon_reaper: {verb} {len(candidates)} over-threshold | "
              f"{len(sub_threshold)} matched-but-young (spared)")
        for c in candidates:
            print(f"  - {verb}: pid {c['pid']} age {c['age_s']}s :: {c['argv']}")
        for c in sub_threshold:
            print(f"  . spared (young): pid {c['pid']} age {c['age_s']}s")
        if leftover and apply:
            print(f"  ! survived: {sorted(leftover)} (WAL truncate may be blocked)")
        if wal is not None:
            print(f"  WAL: {wal}")
    # Exit 2 when something was (or would be) reaped, mirroring pod/worktree audit.
    return 2 if candidates else 0


if __name__ == "__main__":
    sys.exit(main())
```

Notes on load-bearing choices:

- **Match on `args`, not `comm`** (firm). `ps comm` reads `node` / `codex` / `bash` for the daemon trio — the discriminating signal lives in the full argv (`ps -o args`). Confirmed live during planning: `comm` was `node`/`codex`, args carried `codex app-server` / `app-server-broker.mjs serve` / `codex-linux-x64`.
- **`HARD_EXCLUDE` checked FIRST** (firm, R3). During planning the live process table contained (a) the three `codex_task.py --issue 697` review-driver processes, and (b) a `bash -c ... CODEX_COMPANION_SESSION_ID=... eval '...codex-companion...app-server...'` wrapper whose env/eval text literally contains the strings the daemon regexes look for. The `bash -c` exclude + the `codex_task.py` exclude spare both; daemon regexes are anchored (`\bcodex app-server\b`, `app-server-broker\.mjs\b`) so they do not fire on substrings buried in a shell command.
- **`enumerate_candidates(..., snapshot=)`** is injectable so tests pin the predicate against a fake `(pid, etimes, argv)` table with no real processes.
- **`truncate_wal` NEVER VACUUMs** (firm, R5) — VACUUM rewrites the entire 4.4 GB DB; `wal_checkpoint(TRUNCATE)` reclaims only the WAL frames the released connections were pinning, which is the whole win (2.76 GB → 0 in the manual incident).

```
#!/bin/bash
# Recurring Codex app-server daemon reaper — invoked from system crontab.
# The Codex ensemble-review path spawns a persistent daemon trio per session
# that does not exit; they accumulate and pin ~/.codex/logs_2.sqlite (WAL mode)
# open so SQLite can never checkpoint the WAL (it grew to 2.6G by 2026-06-30,
# contributing to a VM root-disk CRITICAL event). This cron reaps daemons older
# than EPS_CODEX_REAPER_MAX_AGE_H (default 24h) and truncates the WAL.
# Mirrors cron_pod_audit.sh / cron_worktree_audit.sh.
# Output: logs/codex_reaper/YYYY-MM-DD.log (one file per day).
set -uo pipefail
export PATH="$HOME/.local/bin:$PATH"
if ! command -v uv >/dev/null 2>&1; then
    echo "$(date -Iseconds) FATAL: uv not on PATH ($PATH); cannot run codex reaper" >&2
    exit 1
fi
PROJECT_DIR="/home/thomasjiralerspong/explore-persona-space"
DATE=$(date +%Y-%m-%d)
LOG_DIR="${EPM_CODEX_REAPER_LOG_DIR:-$PROJECT_DIR/logs/codex_reaper}"
LOG_FILE="$LOG_DIR/$DATE.log"
mkdir -p "$LOG_DIR"
FIRST_RUN_OF_DAY=0
[ -f "$LOG_FILE" ] || FIRST_RUN_OF_DAY=1
{
    echo "=== $(date -Iseconds) codex_reaper start ==="
    cd "$PROJECT_DIR" || exit 1
    uv run python scripts/codex_daemon_reaper.py --apply
    rc=$?
    echo "=== $(date -Iseconds) codex_reaper exit=$rc ==="
} >> "$LOG_FILE" 2>&1
if [ "$FIRST_RUN_OF_DAY" = 1 ]; then
    echo "$(date -Iseconds) codex_reaper: per-pass output → $LOG_FILE (this file receives only this daily pointer line)"
fi
exit 0
```

**Schedule recommendation (negotiable — see §8):** hourly (`17 * * * *`). The leak accrues over days, the threshold is 24h, and an hourly pass is cheap (<1s on a healthy day) and bounds WAL regrowth tightly. Every-15-min is unnecessary (24h threshold); daily would let the WAL regrow a full day between passes. The orchestrator picks the actual cadence at cron registration time.

## 5. Risks and Failure Modes

| # | Risk | Likelihood | Mitigation |
|---|---|---|---|
| R1 | PID-reuse race between selection and signal — a candidate pid recycled by an unrelated process before SIGTERM | Low | `_still_reapable(pid)` re-reads `/proc/<pid>/cmdline` and re-checks match+exclude IMMEDIATELY before every SIGTERM and again before every SIGKILL; a recycled pid fails the re-check and is spared (lifted from `worktree_audit._kill_orphan_pids`). |
| R2 | SIGTERM→SIGKILL wait too short (Node exits via SIGKILL holding the WAL connection → truncate blocked) or too long (burns cron wall time) | Low | `term_wait_s=8.0` (mid-band of the typical 5-10s; `worktree_audit` uses 5s for orphan holders, but Codex Node daemons may take a beat longer to release the sqlite handle on SIGTERM, so 8s buys margin without materially extending a <1s healthy pass). |
| R3 | Selector false positive — args regex matches the `codex_task.py` driver line or a `bash -c` wrapper whose env/eval text contains `codex-companion`/`app-server` (CONFIRMED live during planning) | Medium without guard | `HARD_EXCLUDE` (`codex_task.py`, `workflow_lint`, `wandb_reclaim`, `bash -c`) is checked FIRST, before the daemon match; daemon regexes are word-anchored (`\bcodex app-server\b`, `app-server-broker\.mjs\b`) so they don't fire on substrings inside a shell command. Test (c) + (e) pin this. |
| R4 | A daemon survives SIGKILL → its open connection blocks `wal_checkpoint(TRUNCATE)` | Low | `truncate_wal()` is best-effort: it returns `{ok: False, error: ...}` and reports the surviving pids + the unchanged WAL size in the JSON; it NEVER raises or crashes the reaper. Next pass retries. |
| R5 | New dependency / wrong sqlite handling | Very low | stdlib `sqlite3` only (no new dep; `truncate_wal` NEVER VACUUMs — TRUNCATE reclaims the WAL without rewriting the 4.4 GB DB). Process enumeration is stdlib `ps` + `/proc` (matches `worktree_audit`, no psutil import). |
| R6 | `ps` enumeration fails (cron PATH, container quirk) → reaper silently reaps nothing | Low | `_ps_snapshot()` returns `[]` on `ps` failure → 0 candidates → exit 0 (fail-soft: a reaper that can't read the table does nothing, never mis-kills). The cron's `command -v uv` guard fails loud if `uv` is missing. |
