```python
"""Pins the planner.md §3 Row-coverage authoring guidance (#1208) and asserts
its worked-example declaration lines satisfy the LIVE verify_plan c18 check —
guidance that teaches a non-satisfying shape is worse than none."""

import importlib.util
import re
from pathlib import Path

REPO = Path(__file__).resolve().parents[1]
PLANNER = REPO / ".claude" / "agents" / "planner.md"

TRIGGER_PLAN = (
    "## Hypothesis\n\n"
    "Registered per-row statistic: paired bootstrap difference over 7 pairs.\n\n"
)


def _verify_plan_mod():
    spec = importlib.util.spec_from_file_location(
        "verify_plan_1208", REPO / "scripts" / "verify_plan.py"
    )
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def _example_lines() -> list[str]:
    # The guidance's worked examples are standalone backticked lines
    # starting `Row-coverage` (bullet-indented).
    text = PLANNER.read_text(encoding="utf-8")
    return re.findall(r"(?m)^\s*`(Row-coverage[^`]+)`\s*$", text)


def test_planner_md_has_row_coverage_authoring_guidance():
    text = PLANNER.read_text(encoding="utf-8")
    assert "Registered paired contrast — declare per-arm Row-coverage" in text
    assert "N/A — no paired contrast" in text
    # amendment-standalone sentence present
    assert "verified STANDALONE" in text


def test_guidance_examples_satisfy_c18():
    vp = _verify_plan_mod()
    examples = _example_lines()
    assert len(examples) == 3, examples
    for line in examples:
        plan = TRIGGER_PLAN + line + "\n"
        res = vp.check_paired_contrast_source_coverage(plan, "experiment")
        assert res.status == "PASS", (line, res.detail)  # field name: confirm vs CheckResult


def test_trigger_without_declaration_fails_c18():
    vp = _verify_plan_mod()
    res = vp.check_paired_contrast_source_coverage(TRIGGER_PLAN, "experiment")
    assert res.status == "FAIL", res.detail
```

Notes: the extraction regex deliberately requires the line to START with `Row-coverage` inside backticks, so the byte-exact `N/A — no paired contrast` escape line is NOT extracted as a satisfier example; `len == 3` pins the example count so a dropped/added example is caught. If a future c18 change breaks any example shape, this test fails and forces a guidance redraft in the same round — the drift risk (§8 R1) is thereby mechanical, not aspirational.
