```python
@dataclass(frozen=True)
class StoppingSpec:
    kind: str  # "checkpoint_and_select" | "marker_band_stop" | "fixed_epochs"
    rate_band: tuple[float, float] | None = None        # checkpoint_and_select only
    checkpoint_every_steps: int | None = None
    nat_band: tuple[float, float] | None = None         # marker_band_stop only
    # __post_init__: kind-consistency ValueErrors (never bare assert — behavior.py convention)

@dataclass(frozen=True)
class RecipeSpec:
    behavior_name: str
    overrides: Mapping[str, Any]      # TrainLoraConfig kwargs (read-only copy)
    stopping: StoppingSpec
    generic_frac: float               # fraction of TOTAL training rows that are generic chat
    neg_ratio: float                  # total-negatives per positive; 0.0 == posonly arm
    arm: str                          # in ARMS
    train_method: str                 # in TRAIN_METHODS


def recipe_for(
    behavior: Behavior | str, *, arm: str = "primary",
    generic_frac: float | None = None, train_method: str = "lora",
) -> RecipeSpec:
    b = BEHAVIORS[behavior] if isinstance(behavior, str) else behavior
    if arm not in ARMS: raise ValueError(...)
    if train_method not in TRAIN_METHODS: raise ValueError(...)
    if b.programmatic:
        # Carve-outs: primary lora arm only (parent plan: fullft subset + 3c ablations are content-behavior arms)
        if arm != "primary" or train_method != "lora": raise ValueError(...)
        if generic_frac not in (None, 0.0): raise ValueError("carve-outs take no generic interleave")
        if b.name == "marker":
            return RecipeSpec(b.name, dict(MARKER_OVERRIDES),
                              StoppingSpec("marker_band_stop", nat_band=MARKER_NAT_BAND),
                              generic_frac=0.0, neg_ratio=DEFAULT_NEG_RATIO, arm=arm, train_method="lora")
        if b.name == "taught_fact":
            return RecipeSpec(b.name, dict(FACT_OVERRIDES), StoppingSpec("fixed_epochs"),
                              generic_frac=0.0, neg_ratio=DEFAULT_NEG_RATIO, arm=arm, train_method="lora")
        raise ValueError(f"unknown programmatic behavior {b.name!r}: recipe.py routes only marker/taught_fact")
    gf = DEFAULT_GENERIC_FRAC if generic_frac is None else float(generic_frac)
    if not (0.0 <= gf < 1.0): raise ValueError(...)
    if arm in ("nogeneric", "both_off"): gf = 0.0
    neg = 0.0 if arm in ("posonly", "both_off") else DEFAULT_NEG_RATIO
    # Train-method-HONEST stopping (critic r1 Must-Fix, methodology): the lora path carries
    # the checkpoint ladder; the fullft path CANNOT (train_stage_sft.py hardcodes
    # save_strategy="no"), so its spec says fixed_epochs — honest epoch-grain dosing — instead
    # of advertising a checkpoint_and_select contract the launch command cannot execute.
    # Phase 3b upgrades fullft to a ladder when it patches train_stage_sft.py.
    stopping = (
        StoppingSpec("checkpoint_and_select", rate_band=JUDGED_RATE_BAND,
                     checkpoint_every_steps=CHECKPOINT_EVERY_STEPS)
        if train_method == "lora"
        else StoppingSpec("fixed_epochs")
    )
    return RecipeSpec(b.name, dict(UNIFIED_OVERRIDES), stopping,
                      generic_frac=gf, neg_ratio=neg, arm=arm, train_method=train_method)
```

```python
def build_train_config(
    spec: RecipeSpec, *, run_name: str, seed: int, gpu_id: int = 0,
    tokenizer=None, extra_overrides: Mapping[str, Any] | None = None,
) -> TrainLoraConfig:
    """Materialize the spec as a TrainLoraConfig for train_lora(). Does NOT train.

    tokenizer (optional): when provided and the spec is the marker carve-out, asserts
    tokenizer.encode(MARKER_TEXT, add_special_tokens=False) == [MARKER_TOKEN_ID]
    (marker-leakage-measurement.md in-process assert; #537 [ZLT] no-op incident).
    Phase 0g MUST pass it on the marker path; recipe.py itself stays tokenizer-free.
    """
    if spec.train_method != "lora":
        raise ValueError("fullft specs materialize via fullft_launch_command(), not TrainLoraConfig")
    merged = dict(spec.overrides)
    if extra_overrides:
        illegal = set(extra_overrides) & LOAD_BEARING_KEYS   # the ENUMERATED frozenset in §4.2 (recipe + ladder + marker keys)
        if illegal: raise ValueError(f"extra_overrides may not silently change load-bearing keys: {illegal}")
        merged.update(extra_overrides)
    merged.update({"run_name": run_name, "seed": seed, "gpu_id": gpu_id})
    if spec.behavior_name == "marker" and tokenizer is not None:
        ids = tokenizer.encode(MARKER_TEXT, add_special_tokens=False)
        if ids != [MARKER_TOKEN_ID]: raise ValueError(f"marker tokenization mismatch: {ids}")
    return TrainLoraConfig(**merged)
```

```python
class DoseSelection:
    step: int
    rate: float
    in_band: bool
    fallback: str | None   # None | "closest_approach"

def select_dose_checkpoint(
    rates_by_step: Mapping[int, float], *, band: tuple[float, float] = JUDGED_RATE_BAND,
) -> DoseSelection:
    """Pick the EARLIEST checkpoint whose source judged rate enters [band_lo, band_hi].

    Never the 1.0 ceiling (a ceiling censors the leakage read — #608/#448). If NO
    checkpoint is in band (overshoot between rungs, or never-installed), fall back to
    the closest-approach checkpoint (min distance to the band interval, tie -> earliest)
    with fallback="closest_approach" — the preregistered fallback read mandated by
    marker-training-recipe.md § Multi-arm resolution-band designs item 2. "Never enters
    the band under the recipe" is a reportable outcome, not an infra failure (item 3).

    Critic r1 Must-Fix (statistics): iterate steps in ascending NUMERIC order
    (sorted(rates_by_step)) — insertion order is IGNORED (a checkpoint-* glob yields
    lexical order where "100" < "25"). Band semantics: CLOSED interval [band_lo, band_hi],
    both edges inclusive. Enforced invariant on EVERY returned DoseSelection:
    in_band == (fallback is None) — an out-of-band fallback can never read as band entry.
    Raises ValueError on an empty map or any NaN rate (drop-never-coerce upstream).
    """
```

Pure function ⇒ fully CPU-unit-testable. The judged rates themselves are produced downstream (Phase 0g/1: graded 0–100 Sonnet judge per `llm-judging.md`, binary-rate companion) — this module only encodes the selection rule.
