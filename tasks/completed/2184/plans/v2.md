```python
class RunPodNoPortWedgeError(RunPodError):
    """Raised by :func:`wait_for_ssh` when the pod reached
    ``desired_status=RUNNING`` but exposed no public 22/tcp mapping within the
    bring-up window — the documented RunPod no-port host wedge
    (#664/#770/#1667; provision-time incident #2162 -> #2184).

    Carries the last-observed :class:`PodInfo` (``info``) so callers can read
    ``pod_id`` / ``data_center_id`` for bias-away DC rotation. Subclass of
    :class:`RunPodError` so every existing ``except RunPodError`` caller (the
    provision tail, the resume path) keeps catching it. Deliberately NOT a
    :class:`RunPodNoCapacityError` subclass: the wait-for-capacity retry loop
    must never blind-retry a wedge (same-DC retry is the #2162 failure mode).
    """

    def __init__(self, message: str, info: "PodInfo | None" = None) -> None:
        super().__init__(message)
        self.info = info


class RunPodCpuLaneDryError(RunPodError):
    """CPU-intent residual refusal (#2184): a no-port wedge was detected and
    every rotation candidate DC was dry/wedged (or DC enumeration failed).
    The `cpu_fallback_infeasible_for_plan`-family typed refusal for a fully
    dry/wedged CPU lane; the CLI maps it to EXIT_CPU_LANE_DRY (77) with the
    machine-greppable CPU-LANE-DRY verdict naming the sanctioned GPU-lane
    residual. NOT a RunPodNoCapacityError subclass (must not be wait-retried).
    """
```

**(b) `wait_for_ssh` edit (behavior-preserving except the raise type on the RUNNING shape):**

```python
def wait_for_ssh(pod_id: str, timeout: int = 600, poll_interval: int = 10) -> PodInfo:
    deadline = time.time() + timeout
    info: PodInfo | None = None
    while time.time() < deadline:
        info = get_pod(pod_id)
        if info.ssh_host and info.ssh_port:
            return info
        time.sleep(poll_interval)
    last_status = info.desired_status if info is not None else "unknown"
    msg = (f"Pod {pod_id} did not expose public 22/tcp within {timeout}s. "
           f"Last desiredStatus: {last_status}")
    if info is not None and (info.desired_status or "").upper() == "RUNNING":
        raise RunPodNoPortWedgeError(msg + " (RUNNING-but-no-port wedge, #2184)", info=info)
    raise RunPodError(msg)
```

```python
def _provision_cpu_with_rotation(
    args: argparse.Namespace,
    name: str,
    cpu_instance_id: str,
    intent_label: str,
    *,
    wait_for_capacity: bool,
    explicit_wait_flag: bool,
    data_center_id: str | None,
) -> None:
    """CPU-intent create + bring-up with bias-away DC rotation on a no-port
    wedge (#2184). Attempt 1 preserves today's behavior verbatim (user DC pin
    honored; #2238 wait-for-capacity loop when armed). On a
    RunPodNoPortWedgeError from the provision tail (the wedged pod was already
    auto-terminated by the #2060 path inside the tail), retry create_cpu_pod
    pinned to a DIFFERENT DC — candidates from get_datacenters() minus this
    invocation's wedged DCs minus fresh bad-placement DCs (#2011) — advancing
    on RunPodNoCapacityError (a pinned dry DC is cheap: one GraphQL call, no
    pod). Bounded by EPM_CPU_DC_ROTATION_MAX_WEDGES (default 2 wedge-bearing
    retries) and EPM_CPU_DC_ROTATION_MAX_DCS (default 8 pinned candidates).
    Exhaustion (budget / candidates / enumeration failure) -> the typed
    CPU-LANE-DRY residual refusal (never a silent fallback, never a same-DC

    """
```

Control flow (pseudocode of the loop; the existing #1010 disk-cap clamp block and dry-run early-return run ONCE before it, unchanged):

```python
wedged_dcs: list[str] = []
dry_dcs: list[str] = []
dc_pin = data_center_id            # attempt 1: user pin (or None) — today's behavior
rotation_active = False
candidates: list[str] | None = None   # lazily enumerated on first wedge

while True:
    try:
        if wait_for_capacity and not rotation_active:
            info = create_cpu_pod_with_wait_for_capacity(..., data_center_id=dc_pin)   # unchanged #2238 path
        else:
            info = create_cpu_pod(..., data_center_id=dc_pin)
        print(f"  Created CPU pod {info.pod_id} — waiting for SSH (up to 10 min)...")
        _provision_wait_register_bootstrap(args, name, info, intent_label)
        return
    except WaitForCapacityStillWaiting as exc:
        _emit_still_waiting_and_exit(exc)                       # unchanged
    except RunPodNoPortWedgeError as exc:
        # The tail already recorded the bad placement AND terminated the pod (#2060/#2011).
        wedged = (exc.info.data_center_id if exc.info else None) or dc_pin or "unknown"
        wedged_dcs.append(wedged)
        if len(wedged_dcs) > 1 + _rotation_wedge_budget():      # first wedge + budget retries
            _cpu_lane_dry_refusal(args, intent_label, cpu_instance_id, wedged_dcs, dry_dcs)
        if candidates is None:
            candidates = _rotation_dc_candidates()              # get_datacenters() best-effort; None on failure
        if candidates is None:
            _cpu_lane_dry_refusal(...)                          # enumeration failed — refuse loudly, no rotation
        dc_pin = _next_rotation_dc(candidates, exclude=set(wedged_dcs) | set(dry_dcs))
        if dc_pin is None:
            _cpu_lane_dry_refusal(...)
        rotation_active = True
        print(f"[cpu-dc-rotation] no-port wedge in {wedged}; retrying pinned to {dc_pin} "
              f"({len(wedged_dcs)} wedge(s) so far, budget {_rotation_wedge_budget()}) (#2184)")
    except RunPodNoCapacityError:
        if not rotation_active:
            raise                                               # R5: first-attempt fail-loud UNCHANGED
        dry_dcs.append(dc_pin)
        if len(dry_dcs) >= _rotation_dc_cap():
            _cpu_lane_dry_refusal(...)
        dc_pin = _next_rotation_dc(candidates, exclude=set(wedged_dcs) | set(dry_dcs))
        if dc_pin is None:
            _cpu_lane_dry_refusal(...)
        print(f"[cpu-dc-rotation] {dry_dcs[-1]} dry (no capacity); trying {dc_pin} (#2184)")
```

- `_rotation_dc_candidates() -> list[str] | None`: `[d["id"] for d in get_datacenters() if d.get("id")]` under `try/except Exception` → WARN + `None` (mirrors `_different_dc_hint`'s degrade at `:1874-1892`); additionally drops DCs carried by `fresh_bad_hosts()` entries' `dc_id` fields (the #2011 cross-invocation bias-away memory).
