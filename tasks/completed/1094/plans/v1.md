### 4.1 New code in `scripts/verify_plan.py` (inserted after c26, line ~3963, before the Driver section)

Module-level regexes (file convention: defined adjacent to the check with grounding comments):

```python
# ─── Check 27 — decision-band precedent coherence (WARN-only) ───────────────

# Band line: a non-fenced, bold-labeled list item inside a decision-keyword
# section (_C13_GATE_SECTION_RE reused) carrying a multiplicative threshold
# `cmp T ×` with fractional T (0 < T <= 1) — the #931 committed-threshold
# idiom ("< 0.5 × 0.588", "≥ 0.5× its ceiling"). Integer/super-unity T ("≥ 2×
# wall-time" kill fences) deliberately excluded (fraction-of-ceiling bands
# are the target class; wall-time multipliers are a different quantity).
_C27_BAND_RE = re.compile(r"(?P<cmp><=|>=|[<>≤≥])\s*(?P<thr>0?\.\d+|1\.0|1)\s*[×x](?![a-zA-Z])")
_C27_LIST_ITEM_RE = re.compile(r"^\s{0,3}(?:[-*]|\d+\.)\s")          # c14 sibling
_C27_BOLD_LABEL_RE = re.compile(r"\*\*([^*\n]{1,60})\*\*")           # c14 sibling

# Precedent-ratio assertion: explicit `ratio ≈ r` / `ratio ≈ r1–r2` token.
# Decimal point REQUIRED (excludes the `ratio ~1:1` mix idiom — the only 2
# non-incident same-line corpus hits); `%`-suffixed ratios NOT harvested
# (percent-vs-fraction confusion is a named FP mode — accepted false neg).
_C27_RATIO_RE = re.compile(
    r"(?i)\bratios?\s*[≈=~]\s*(?P<r1>0?\.\d+)(?:\s*[–—-]\s*(?P<r2>0?\.\d+))?(?!\s*%)"
)
# Verb-anchored side vocabulary (navigation "see below / table below" cannot
# match: a verb is required). Negation guard drops "not/never well below".
_C27_BELOW_RE = re.compile(r"(?i)\b(?:well|lands?|sits?|stays?|falls?)\s+below\b|\bunder\s+half\b")
_C27_ABOVE_RE = re.compile(r"(?i)\bexceeds?\b|\b(?:well|lands?|sits?|stays?)\s+above\b|\bat\s+least\s+half\b")
_C27_NEG_RE = re.compile(r"(?i)\b(?:not|never|no\s+longer)\s+(?:(?:well|lands?|sits?|stays?|falls?)\s+)?(?:below|above)\b|\bexcept\s")
# Same-line recompute corroborator: positive decimals split at the first
# `vs`; slash (`/`) is NOT a ratio separator in this corpus (it is the
# paired-cells idiom: "rotated +0.349/+0.334").
_C27_VS_RE = re.compile(r"(?i)\bvs\.?\s")
_C27_POSNUM_RE = re.compile(r"(?<![\d.\-])\+?(0?\.\d{2,4})\b")
```

Functions:

```python
def _c27_bands(plan: str) -> list[dict]:
    """[{label, cmp, thr: Fraction, line}] — mirror of _c13_registered_gates'
    fence-masked, section-scoped walk (lines 1369-1377), with the band
    predicate: list item + bold label + _C27_BAND_RE hit, thr in (0, 1]."""

def _c27_ratio_assertions(plan: str) -> list[dict]:
    """[{line, side: 'below'|'above', candidates: set[Fraction]}] over
    non-fenced lines. Fire only when a _C27_RATIO_RE hit AND exactly one
    side (below XOR above vocab, _C27_NEG_RE-guarded) co-occur on the line.
    candidates = quoted r1 (and r2 if range) UNION same-line vs-pair
    recompute: numerators = positive decimals left of the first `vs`
    (ratio-token spans blanked first), denominator = FIRST positive decimal
    right of `vs` (ratio spans blanked); keep a/b only when b > 0 and
    0 < a/b <= 2 (sanity window). Fraction arithmetic throughout — exact
    boundary semantics at r == T, no float-equality wobble."""

def _c27_na_escape_declared(plan: str) -> bool:
    return _standalone_na_declared(plan, r"no precedent[- ]labeled decision bands?")

def check_precedent_band_coherence(plan: str, kind: str) -> CheckResult:
    """WARN-only, conditional: a registered fractional decision band (cmp T ×
    in a success/kill/decision section), applied to a same-line side-asserted
    precedent ratio the plan itself quotes/implies, must land in the branch
    the narrative asserts. Straddle (a quoted range containing both sides of
    T while one side is asserted) also WARNs. NEVER FAILs (c14 doctrine);
    the semantic verdict stays with the Statistics critic (lens item 3).
    Incident: #825 v17 (0.5× band vs instruct precedent 0.3489/0.6731 =
    0.519, narrated 'well below'; caught only by the Codex critic)."""
    cid, name = "c27_precedent_band_coherence", "decision-band precedent coherence"
    if kind not in ("experiment", "analysis"):
        return _skip(cid, name, "kind-exempt: precedent-labeled decision bands are an "
                                "experiment|analysis plan shape")
    bands = _c27_bands(plan)
    if not bands:
        return _skip(cid, name, "no registered multiplicative decision band detected")
    if _c27_na_escape_declared(plan):
        return _pass(cid, name, "explicit N/A declared (no precedent-labeled decision bands)")
    thresholds = {b["thr"] for b in bands}
    if len(thresholds) != 1:
        return _skip(cid, name, f"{len(thresholds)} distinct band thresholds — "
                                "precedent-to-band pairing ambiguous at the plan surface")
    T = next(iter(thresholds))
    assertions = _c27_ratio_assertions(plan)
    if not assertions:
        return _skip(cid, name, "band present but no side-asserted precedent ratio line detected")
    offenders = []
    for a in assertions:
        lo, hi = min(a["candidates"]), max(a["candidates"])
        if a["side"] == "below" and hi >= T:
            offenders.append((a, "contradiction" if lo >= T else "straddle"))
        elif a["side"] == "above" and lo < T:
            offenders.append((a, "contradiction" if hi < T else "straddle"))
    if not offenders:
        return _pass(cid, name, f"{len(assertions)} side-asserted precedent ratio line(s) "
                                f"coherent with the registered {float(T):g}× band")
    return _warn(cid, name, _c27_offender_detail(offenders, T, bands))
```

Registration: append `check_precedent_band_coherence` to `CHECKS` after `check_gpu_basis_routed_machine` (line 3992).

## 7. Risks and failure modes

| Risk | Likelihood | Mitigation |
|---|---|---|
| FP: ratio line about a DIFFERENT quantity than the band's (mis-pairing) | Low–Med | Single-distinct-threshold gate (multi-T → SKIP); WARN severity; N/A escape; corpus-sweep bound |
| FP: percent-vs-fraction / mix-ratio (`~1:1`) confusion | Low | `%` not harvested; decimal-point required (corpus-verified: kills both observed idiom hits) |
